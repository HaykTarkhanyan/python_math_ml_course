% ============================================================ SECTION 1
\section{A head is two circuits}
\sectiontransition{A head is two circuits}{Where to look, and what to move - two different matrices}

\begin{frame}{The attention head, written out}
  \vspace{0.3em}
  From the attention lecture, for a residual-stream row vector $x_i$ at each position $i$:

  \vspace{0.3em}
  \[
    q_i = x_i W_Q, \qquad k_j = x_j W_K, \qquad v_j = x_j W_V
  \]
  \[
    \text{score}_{ij} = q_i \cdot k_j, \qquad a_{ij} = \operatorname{softmax}_j(\text{score}_{ij}),
    \qquad \text{out}_i = \Big(\sum_j a_{ij}\, v_j\Big) W_O
  \]

  \vspace{0.4em}
  Four matrices: $W_Q, W_K, W_V$ are $512 \times 64$, $W_O$ is $64 \times 512$.

  \vspace{0.6em}
  \keybox{\small The trick of this lecture is to stop thinking of four matrices and regroup them
  into \textbf{two} - one for each question the head answers.}
\end{frame}

\begin{frame}{Where to look: the QK circuit}
  \vspace{0.3em}
  Write the score out, one step at a time:
  \begin{align*}
    \text{score}_{ij}
      &= q_i\, k_j^{\top} && \text{dot product as a row times a column}\\
      &= (x_i W_Q)(x_j W_K)^{\top} && \text{substitute } q, k\\
      &= x_i\, W_Q W_K^{\top}\, x_j^{\top} && (AB)^{\top} = B^{\top}A^{\top}\\
      &= x_i\, W_{QK}\, x_j^{\top}, && W_{QK} := W_Q W_K^{\top} \;\;(512 \times 512)
  \end{align*}

  \vspace{0.2em}
  \keybox{\small $W_{QK}$ - the \textbf{query-key} (QK) circuit - alone decides \textbf{which positions
  attend to which}. $W_Q$ and $W_K$
  never appear separately in anything the head does - only their product.}
\end{frame}

\begin{frame}{What to move: the OV circuit}
  \vspace{0.3em}
  Now the output, the same way:
  \begin{align*}
    \text{out}_i
      &= \Big(\sum_j a_{ij}\, x_j W_V\Big) W_O && \text{substitute } v_j\\
      &= \sum_j a_{ij}\, x_j\, W_V W_O && \text{a sum times a matrix}\\
      &= \sum_j a_{ij}\, x_j\, W_{OV}, && W_{OV} := W_V W_O \;\;(512 \times 512)
  \end{align*}

  \vspace{0.3em}
  \keybox{\small $W_{OV}$ - the \textbf{output-value} (OV) circuit - alone decides \textbf{what gets
  written} once the head has looked. The
  attention pattern $a_{ij}$ only chooses \emph{how much} of each position to move.}

  \vspace{0.2em}
  {\footnotesize Both matrices are $512 \times 512$ but have rank at most 64: a head can only read
  and write a 64-dimensional slice of the stream.}
\end{frame}

\begin{frame}{By hand: a head in a two-number world}
  \vspace{0.1em}
  Three tokens with 2-number embeddings: $A = (1, 0)$, $B = (0, 1)$, $C = (0.5, -0.5)$. The
  current token is $A$; the earlier tokens are $A$, $B$, $C$.

  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small
      $W_{QK} = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix}$, so $A\,W_{QK} = (0, 1)$\\[0.4em]
      scores: $A{:}\ 0$, \; $B{:}\ 1$, \; $C{:}\ -0.5$\\[0.2em]
      softmax: $0.23,\; \mathbf{0.63},\; 0.14$\\[0.2em]
      \textcolor{popblue}{$A$ mostly looks at $B$.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \small
      $W_{OV} = I$ (copy what you see):\\
      out $= 0.23A + 0.63B + 0.14C = (0.30,\; 0.56)$\\[0.2em]
      read out: $A{:}\ 0.30$, \; $\mathbf{B{:}\ 0.56}$, \; $C{:}\ -0.13$\\[0.4em]
      $W_{OV} = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix}$ (swap):\\
      out $= (0.56,\; 0.30)$ $\rightarrow$ now it predicts $\mathbf{A}$.
    \end{column}
  \end{columns}

  \vspace{0.6em}
  \keybox{\small Same $W_{QK}$, same attention, two different predictions. \textbf{Where it looks}
  and \textbf{what it writes} really are separate - the warning from \emph{Opening the box}, now as
  two matrices.}
\end{frame}

\begin{frame}{Reading the circuits in token space}
  \vspace{0.4em}
  A 512-dimensional matrix is still unreadable. Sandwich it between the embedding $W_E$ (token
  $\rightarrow$ stream) and the unembedding $W_U$ (stream $\rightarrow$ logits):

  \vspace{0.5em}
  \begin{center}
  \small
  \begin{tabular}{@{}lll@{}}
    \toprule
    & \textbf{matrix} & \textbf{entry $(s, t)$ means}\\
    \midrule
    QK in token space & $W_E\, W_{QK}\, W_E^{\top}$ & how much token $s$ wants to attend to token $t$\\[0.3em]
    OV in token space & $W_E\, W_{OV}\, W_U$ & what attending to token $s$ does to the logit of $t$\\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.6em}
  Both are $48{,}262 \times 48{,}262$ - over two billion numbers. Nobody looks at them. But we can
  \textbf{ask them questions}: is the diagonal large? which rows stand out? That is the rest of the
  lecture.
\end{frame}

% ============================================================ SECTION 2
\section{Zero layers: a bigram table}
\sectiontransition{Zero layers: a bigram table}{What can a model predict with no attention at all?}

\begin{frame}{The direct path}
  \vspace{0.6em}
  Remove every head. The stream at the last position is just the current token's embedding, and
  the prediction is
  \[
    \text{logits} \;=\; \operatorname{LayerNorm}(W_E[\text{token}])\; W_U .
  \]

  \vspace{0.3em}
  The next token depends on the current token and \textbf{nothing else}. That is a
  \textbf{bigram table}: ``after this word, that word'' - the simplest language model there is.

  \vspace{0.6em}
  \keybox{\small Every transformer contains this path, because the residual stream carries the
  embedding all the way to the end. In the one-layer model it is one term of the sum, and we can
  read it straight off $W_E$ and $W_U$.}
\end{frame}

\begin{frame}{Reading the bigram table}
  \vspace{0.3em}
  The one-layer model's direct path, top guesses for a few tokens:

  \vspace{0.4em}
  \begin{center}
  \small
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{current token} & \textbf{top next token} & \textbf{runner-up}\\
    \midrule
    ``\,United'' & \textcolor{paramgreen}{``\,States''} 99.8\% & ``\,Kingdom'' 0.1\%\\
    ``\,Hong'' & \textcolor{paramgreen}{``\,Kong''} 96\% & ``s'' 2\%\\
    ``\,according'' & \textcolor{paramgreen}{``\,to''} 99.5\% & ``\,same'' 0.3\%\\
    ``\,New'' & \textcolor{paramgreen}{``\,York''} 71\% & ``\,Zealand'' 24\%\\
    ``\,San'' & \textcolor{paramgreen}{``\,Francisco''} 57\% & ``itation'' 17\%\\
    ``\,Barack'' & \textcolor{armred}{``\,and''} 43\% & ``able'' 6\%\\
    ``\,import'' & \textcolor{armred}{``ation''} 88\% & ``able'' 8\%\\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.4em}
  \watchbox{\small The last rows are kept on purpose. The direct path is \emph{one term} of the
  model's sum - the attention layer adds its own contribution on top, and the model as a whole
  does not have to get every bigram right through this path alone.}
\end{frame}

% ============================================================ SECTION 3
\section{One layer: copying}
\sectiontransition{One layer: copying}{What eight heads can do, and the one thing they cannot}

\begin{frame}{What one layer adds}
  \vspace{0.4em}
  With one attention layer the logits are the direct path plus one term per head:
  \[
    \text{logits} \;=\; \underbrace{W_E\,W_U}_{\text{bigram}} \;+\;
    \sum_{\text{heads}}\; \underbrace{\textstyle\sum_j a_{ij}}_{\text{QK: where}}\;
    \underbrace{W_E\,W_{OV}\,W_U}_{\text{OV: what}}[\text{token at } j]
  \]

  \vspace{0.4em}
  So a head at the current position looks back at some earlier token $s$ and adds row $s$ of its
  OV table to the prediction. Three tokens are involved: the earlier one, the current one, and the
  predicted one - the paper calls these \textbf{skip-trigrams}: ``$s$ \ldots\ current
  $\rightarrow$ next''.

  \vspace{0.5em}
  \keybox{\small The most useful skip-trigram of all is also the simplest: \emph{whatever earlier
  word I look at, predict it again}. Text repeats itself - names, variables, topics. A head that
  does this is a \textbf{copying head}.}
\end{frame}

\begin{frame}{How to check a two-billion-entry table}
  \vspace{0.3em}
  A copying head's OV table $W_E W_{OV} W_U$ has a big diagonal: token $t$ maps to token $t$.
  Two ways to test it without printing it:

  \vspace{0.5em}
  \textbf{1. Directly.} Take 2000 random tokens, compute each one's row, and ask whether the
  token itself is the top entry.

  \vspace{0.5em}
  \textbf{2. With eigenvalues.} A matrix that maps things to themselves has mostly
  \emph{positive} eigenvalues. And we never need the big matrix:
  \[
    \text{the nonzero eigenvalues of } \underbrace{W_E\, W_{OV}\, W_U}_{48{,}262 \times 48{,}262}
    \;=\; \text{those of } \underbrace{W_{OV}\, W_U\, W_E}_{512 \times 512}
  \]
  because $ABC$ and $BCA$ always share their nonzero eigenvalues. Score
  $= \sum \lambda / \sum |\lambda|$: $+1$ means all positive, $-1$ all negative.
\end{frame}

\end{frame}

\begin{frame}{Why one layer cannot do induction}
  \vspace{0.3em}
  To continue ``\ldots\ A B \ldots\ A'' with B, the head at the second A must attend to the
  position \textbf{right after} the first A.

  \vspace{0.5em}
  In one layer, a key at position $j$ is built from the token at $j$ (and its position) - nothing
  else. Nothing at the position of B says ``the token before me was A''.

  \vspace{0.6em}
  \trapbox{\small So the head cannot find B by looking for A. The best it can do is copy \emph{every}
  earlier token a little - which is what the measurement showed: the right token's median rank
  improves from 18{,}122 to 15{,}386, and it is top-1 exactly \textbf{0\%} of the time.}

  \vspace{0.4em}
  \keybox{\small Some \textbf{earlier} head has to write ``the token before me was A'' into position
  B's stream first. That needs a second layer.}
\end{frame}

% ============================================================ SECTION 4
\section{Two layers: composition}
\sectiontransition{Two layers: composition}{A head in layer 1 can read what a head in layer 0 wrote}

\begin{frame}{Three ways to compose}
  \vspace{0.3em}
  A layer-1 head reads the stream, and the stream now contains layer-0 heads' outputs. Which of
  its inputs does it build from them?

  \vspace{0.5em}
  \begin{center}
  \small
  \begin{tabular}{@{}p{0.2\textwidth}p{0.36\textwidth}p{0.32\textwidth}@{}}
    \toprule
    & \textbf{the layer-1 head's \ldots\ uses layer-0 output} & \textbf{matrix to check}\\
    \midrule
    Q-composition & query - \emph{what it is looking for} & $W_{OV}^{(0)}\, W_{QK}^{(1)}$\\[0.3em]
    K-composition & key - \emph{what it finds} & $W_{QK}^{(1)}\, W_{OV}^{(0)\top}$\\[0.3em]
    V-composition & value - \emph{what it moves} & $W_{OV}^{(0)}\, W_{OV}^{(1)}$\\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.5em}
  \keybox{\small Induction needs \textbf{K-composition}: the induction head's \emph{keys} must carry
  ``the token before me was A'', which only a layer-0 head can have written.}
\end{frame}

\begin{frame}{A composition score}
  \vspace{0.5em}
  How much does head $h_1$ read from what head $h_0$ writes? For K-composition:
  \[
    \text{score}(h_0 \rightarrow h_1) \;=\;
    \frac{\big\| W_{QK}^{(h_1)}\, W_{OV}^{(h_0)\top} \big\|_F}
         {\big\| W_{QK}^{(h_1)} \big\|_F\; \big\| W_{OV}^{(h_0)} \big\|_F}
  \]

  \vspace{0.3em}
  \begin{itemize}\small
    \item $\|\cdot\|_F$ is the Frobenius norm - the square root of the sum of all squared entries.
    \item The ratio is between 0 and 1: near 0 when $h_0$ writes into directions $h_1$ ignores,
      larger when they overlap.
    \item Random matrices of the same shape score about \textbf{0.044} - the baseline to beat.
  \end{itemize}

  \vspace{0.5em}
  \watchbox{\small It is a \textbf{norm}, computed from weights. A large score says the two heads
  \emph{could} talk; it does not say they do, on any particular input.}
\end{frame}

\begin{frame}{Reading the induction head off the weights}
  \vspace{0.3em}
  Follow one key through both layers:

  \vspace{0.3em}
  \begin{enumerate}
    \item At position $j$, the previous-token head L0H3 attends to position $j-1$ and writes that
      token through its OV circuit: $\;\text{key input} = W_E[\text{token}_{j-1}]\, W_{OV}^{\text{L0H3}}$.
    \item At the current position, the induction head's query is built from the current token:
      $W_E[\text{token}_i]$.
    \item Its score for position $j$ is therefore
      \[
        W_E[\text{token}_i]\;\; W_{QK}^{\text{L1H6}}\;\; \big(W_E[\text{token}_{j-1}]\, W_{OV}^{\text{L0H3}}\big)^{\top}
      \]
  \end{enumerate}

  \vspace{0.3em}
  \keybox{\small A token $\times$ token table: rows are the current token, columns are the token
  \emph{before} a key. If this is an induction head, the table should be largest where the two are
  the \textbf{same} token - on the diagonal.}
\end{frame}

  \end{tikzpicture}
  \end{center}

  \vspace{1.0em}
  \keybox{\small This is why induction needs two layers: the second head's \emph{key} is built out
  of the first head's \emph{output}. One layer has nothing to build it from.}
\end{frame}

% ============================================================ SECTION 6
\section{What weights can and cannot say}
\sectiontransition{What weights can and cannot say}{Exact and data-free - up to the first MLP}

\begin{frame}{The strengths}
  \vspace{0.6em}
  \begin{itemize}
    \item \textbf{No data needed.} The composition score and the diagonal test ran on no text at
      all. What you find holds for \emph{every} input, not the ones you happened to try.
    \item \textbf{Exact, in attention-only models.} The logits really are sums of products of these
      matrices - nothing is approximated except LayerNorm's scale.
    \item \textbf{It explains, not only locates.} Running the model told us \emph{that} two layers
      are needed. The weights told us \emph{why}: a key built from another head's output.
  \end{itemize}

  \vspace{0.6em}
  \keybox{\small This is how induction heads were first understood - on models like the two we used
  today - before anyone went looking for them in large models.}
\end{frame}

\begin{frame}{The limits}
  \vspace{0.4em}
  \begin{itemize}
    \item \textbf{MLPs break it.} A nonlinearity between two heads means their combined effect is no
      longer a product of matrices. Real models have an MLP in every block.
    \item \textbf{Paths multiply.} Two heads give pairs; three give triples. In GPT-2 small there are
      144 heads and every later head can compose with every earlier one.
    \item \textbf{Scores are norms, not effects.} A large composition score says two heads
      \emph{could} talk - L4H7 is the proof that it does not mean they do.
    \item \textbf{Superposition.} A token's embedding is not a clean input when the stream packs
      many features into each direction - the subject of the features lecture.
  \end{itemize}

  \vspace{0.4em}
  \trapbox{\small So for real models the field moved from reading weights to \textbf{intervening on
  activations}. Weights give you candidates; interventions decide.}
\end{frame}

% ============================================================ RECAP
\begin{frame}{Recap}
  \vspace{0.3em}
  \begin{itemize}
    \item A head is two independent circuits: \textbf{QK} decides where to look, \textbf{OV} decides
      what to write.
    \item No layers: a \textbf{bigram table}, read from $W_E W_U$.
    \item One layer: \textbf{copying} heads - checked directly and with eigenvalues, which disagree
      on three heads.
    \item Two layers: \textbf{composition}. The previous-token $\rightarrow$ induction pair is the top
      K-composition of 64; the induction head's search rule and copying are both readable from the
      weights.
    \item GPT-2 small: the weights point at the right partner, by a hair.
  \end{itemize}

  \vspace{0.4em}
  \takebox{\small Reading weights explains \emph{why} a circuit works, for every input at once - in
  models small and linear enough to allow it.}
\end{frame}

\begin{frame}{Next}
  \vspace{1.2em}
  \takebox{\textbf{Next core lecture: does it actually do that?}\\[0.6em]
  We stop reading and start \textbf{intervening}: delete a head, paste an activation from one run
  into another, and measure what changes. It is the method that works on real models - and it is
  how you would settle L4H11 against L4H7.}
\end{frame}
