% Cold open
\begin{frame}{Every model so far assumed the rows were interchangeable}
  Regression, trees, logistic regression -- all of them treat the dataset as a
  \emph{bag} of i.i.d. rows. Shuffle the rows, nothing changes.
  \vfill
  \callout{armred}{\textbf{Time series breaks that assumption.}
  Row order \emph{is} the signal. Yesterday's value tells you a lot about
  today's. You can only train on the past, and you must predict the future --
  never the other way around.}
  \vfill
  Examples everywhere: daily sales, electricity demand, website traffic, the
  dram-to-dollar exchange rate, sensor readings, COVID case counts, your heart rate.
  \vfill
  \footnotesize Two lectures. \textbf{Today (30):} the classical statistical
  toolkit -- decomposition, stationarity, ARIMA, exponential smoothing.
  \textbf{Next (31):} treating forecasting as a supervised ML problem.
\end{frame}

\section{What makes time series different}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{What makes time series different}}\\[4pt]
    \small One column, indexed by time -- and the index carries information.
  \end{center}
\end{frame}

\begin{frame}{The vocabulary}
  A time series is a sequence $y_1, y_2, \dots, y_T$ sampled at regular steps
  (hourly, daily, monthly, \dots).
  \vfill
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Two jobs:}
    \begin{itemize}\small
      \item \textbf{Forecasting} -- predict future values $y_{T+1}, \dots, y_{T+h}$.
        ($h$ = forecast \emph{horizon}.)
      \item \textbf{Understanding} -- decompose, explain, find structure.
    \end{itemize}
    \vskip6pt
    \textbf{Two shapes of data:}
    \begin{itemize}\small
      \item \textbf{Univariate} -- one series.
      \item \textbf{Multivariate} -- many series, possibly with exogenous drivers.
    \end{itemize}
    \column{0.5\textwidth}
    \callout{armblue}{\footnotesize
      \textbf{The one rule that governs everything:}\\
      information flows forward in time only. Any value, feature, or statistic you
      feed the model must have been \emph{knowable at prediction time}. Violate this
      and you get ``leakage'' -- great backtests, useless in production.}
  \end{columns}
\end{frame}

\begin{frame}{The four components of a series}
  \begin{columns}[T]
    \column{0.44\textwidth}
    A series is usefully thought of as a sum (or product) of:
    \begin{itemize}\small
      \item \textbf{Trend} $T_t$ -- long-run drift up or down.
      \item \textbf{Seasonality} $S_t$ -- a pattern that repeats on a \emph{fixed}
        period (24h, 7d, 12mo).
      \item \textbf{Cycle} -- swings with no fixed period (business cycles).
      \item \textbf{Noise / residual} $R_t$ -- what's left.
    \end{itemize}
    \vskip4pt
    \footnotesize Additive: $y_t = T_t + S_t + R_t$. \\
    Multiplicative: $y_t = T_t \cdot S_t \cdot R_t$ (season grows with level --
    take $\log$ to make it additive).
    \column{0.56\textwidth}
    \includegraphics[width=\textwidth]{fig/ts_anatomy.pdf}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Decomposition: STL}
  \begin{columns}[T]
    \column{0.44\textwidth}
    \textbf{STL} = ``Seasonal-Trend decomposition using Loess''. It splits the
    series into trend + seasonal + residual, robustly.
    \vskip6pt
    Why bother?
    \begin{itemize}\small
      \item \emph{See} the structure before modeling.
      \item Deseasonalize (analyze the trend cleanly).
      \item The residual is where anomalies show up.
    \end{itemize}
    \vskip6pt
    {\lstset{basicstyle=\ttfamily\scriptsize}\begin{lstlisting}
from statsmodels.tsa.seasonal import STL
res = STL(y, period=12).fit()
res.trend, res.seasonal, res.resid
\end{lstlisting}}
    \column{0.56\textwidth}
    \includegraphics[width=\textwidth]{fig/stl_decomposition.pdf}
  \end{columns}
\end{frame}

\section{Stationarity and differencing}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Stationarity and differencing}}\\[4pt]
    \small The classical models need a series whose behaviour does not drift.
  \end{center}
\end{frame}

\begin{frame}{What is stationarity?}
  \begin{columns}[T]
    \column{0.44\textwidth}
    A series is (weakly) \textbf{stationary} if its statistical behaviour does not
    depend on \emph{when} you look:
    \begin{itemize}\small
      \item constant mean (no trend),
      \item constant variance (no growing swings),
      \item autocovariance depends only on the \emph{lag}, not on absolute time.
    \end{itemize}
    \vskip4pt
    \callout{armorange}{\scriptsize Why we care: ARMA-type models assume
    stationarity. Trends and seasonality must be removed \emph{first}, then added
    back to the forecast.}
    \column{0.56\textwidth}
    \includegraphics[width=\textwidth]{fig/stationarity.pdf}
  \end{columns}
\end{frame}

\begin{frame}{Differencing makes a series stationary}
  \begin{columns}[T]
    \column{0.42\textwidth}
    \textbf{Differencing:} model the change instead of the level.
    \[ y'_t = y_t - y_{t-1} \]
    \begin{itemize}\small
      \item Removes a trend (do it $d$ times -- usually $d=1$).
      \item \textbf{Seasonal differencing} $y_t - y_{t-m}$ removes a season of
        period $m$.
    \end{itemize}
    \vskip4pt
    \textbf{Testing it -- two tests, opposite nulls} {\small(small $p$ rejects, so
    they point opposite ways):}
    \begin{itemize}\scriptsize
      \item \textbf{Augmented Dickey-Fuller} (\textbf{ADF}) -- $H_0$: non-stationary.
      \item \textbf{Kwiatkowski-Phillips-Schmidt-Shin} (\textbf{KPSS}) -- $H_0$:
        stationary.
    \end{itemize}
    \column{0.58\textwidth}
    \includegraphics[width=\textwidth]{fig/differencing.pdf}
  \end{columns}
\end{frame}

\begin{frame}{Predict-first: if one difference is good, are two better?}
  One difference took us from ADF $p = 0.94$ to $p = 0.000$. The series looks
  stationary. Tempting thought: difference once more, just to be safe.
  \vfill
  \textbf{Predict: what happens to the spread of the values?}
  \pause
  \vfill
  \begin{center}
    \includegraphics[width=0.86\textwidth]{fig/over_differencing.pdf}
  \end{center}
  \vskip1pt
  \callout{armred}{\footnotesize The spread goes \emph{up}: $40.9 \to 10.8 \to 11.7$.
  No trend was left to remove, so the second difference only differenced the
  \textbf{noise} -- and the difference of noise is noisier still.}
\end{frame}

\section{Autocorrelation: ACF and PACF}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Autocorrelation}}\\[4pt]
    \small How a series correlates with delayed copies of itself.
  \end{center}
\end{frame}

\begin{frame}{ACF and PACF}
  \begin{columns}[T]
    \column{0.42\textwidth}
    \textbf{ACF} (autocorrelation): correlation between $y_t$ and $y_{t-k}$ for
    each lag $k$. Includes indirect effects.
    \vskip4pt
    \textbf{PACF} (partial): correlation at lag $k$ \emph{after} removing the
    effect of the shorter lags. The direct link only.
    \vskip8pt
    They are the classic way to read off model orders:
    \begin{itemize}\scriptsize
      \item PACF cuts off after lag $p$ $\Rightarrow$ \textbf{AR($p$)}.
      \item ACF cuts off after lag $q$ $\Rightarrow$ \textbf{MA($q$)}.
      \item Both decay $\Rightarrow$ mix (ARMA).
    \end{itemize}
    \column{0.58\textwidth}
    \includegraphics[width=\textwidth]{fig/acf_pacf.pdf}
    \vskip2pt
    \footnotesize The shaded band is the ``not significantly different from
    zero'' region ($\pm 1.96/\sqrt{T}$).
  \end{columns}
\end{frame}

\section{ARIMA family}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{The ARIMA family}}\\[4pt]
    \small The workhorse of classical forecasting.
  \end{center}
\end{frame}

\begin{frame}{Building up: AR, MA, ARMA}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{AR($p$) -- autoregressive.} Regress on your own past:
    \[ y_t = c + \sum_{i=1}^{p}\phi_i\, y_{t-i} + \eps_t \]
    {\footnotesize ``Tomorrow looks like a weighted blend of recent days.''}
    \vskip2pt
    \textbf{MA($q$) -- moving average.} Regress on past \emph{shocks}:
    \[ y_t = c + \eps_t + \sum_{j=1}^{q}\theta_j\, \eps_{t-j} \]
    {\footnotesize ``A surprise last week still echoes this week.''}
    \column{0.5\textwidth}
    \callout{armblue}{\textbf{ARMA($p,q$)} combines both -- for a
    \emph{stationary} series.}
    \vskip3pt
    \textbf{ARIMA($p,d,q$)} adds the ``I'' = Integrated = difference $d$ times
    first, so it handles trending series directly.
    \begin{itemize}\scriptsize
      \item $p$ -- AR order (from PACF)
      \item $d$ -- differencing (from ADF)
      \item $q$ -- MA order (from ACF)
    \end{itemize}
    \footnotesize ARIMA(0,1,0) is just the random walk -- the ``tomorrow $=$
    today'' baseline.
  \end{columns}
\end{frame}

\begin{frame}{What AR and MA actually look like}
  \begin{center}
    \includegraphics[width=0.96\textwidth]{fig/ar_vs_ma.pdf}
  \end{center}
  \vskip 4pt
  \callout{armblue}{\footnotesize Same machinery, different memory: an \textbf{AR} term makes
  the series \emph{persist} (today leans on yesterday's \emph{value}); an \textbf{MA} term is
  a short \emph{echo} of the last random \emph{shock}. ARMA blends the two.}
\end{frame}

\begin{frame}{Careful: ``moving average'' means two different things}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \callout{armred}{\textbf{The MA in ARIMA is \emph{not} a rolling mean.}}
    \vskip6pt
    \textbf{What you already know} -- a \emph{rolling} or \emph{moving} average: a
    sliding mean of the \textbf{values}, used to smooth a curve.
    \[ \tfrac{1}{3}\big(y_{t-1}+y_{t-2}+y_{t-3}\big) \]
    You compute it \emph{from data you can see}.
    \column{0.52\textwidth}
    \textbf{MA($q$) in ARIMA} -- a weighted sum of the past \textbf{shocks}
    (the model's own errors):
    \[ y_t = c + \eps_t + \theta_1\eps_{t-1} + \dots \]
    The $\eps$'s are \emph{unobserved}. You never measure them; the fitting procedure
    infers them. That is why MA orders cannot be read off the data directly the way a
    rolling mean can -- and why they need the ACF.
  \end{columns}
  \vfill
  \callout{armorange}{\footnotesize Two unrelated ideas, one name, purely for historical
  reasons. Next lecture we build rolling-mean \emph{features}
  (\texttt{y.shift(1).rolling(7).mean()}) and feed them to a tree -- that is the
  \emph{other} moving average. Keep them apart.}
\end{frame}

\begin{frame}{SARIMA: adding seasonality}
  Real series are seasonal, so add a second, seasonal triple:
  \[ \text{SARIMA}(p,d,q)(P,D,Q)_m \]
  \begin{itemize}\small
    \item lower-case $(p,d,q)$ -- the short-term, non-seasonal part;
    \item upper-case $(P,D,Q)$ -- the same idea at the seasonal lag;
    \item $m$ -- periods per season (12 monthly, 7 daily-with-weekly, 24 hourly-with-daily).
  \end{itemize}
  \vfill
  \callout{armorange}{\footnotesize One $m$ only. Hourly electricity demand has a daily
  \emph{and} a weekly \emph{and} a yearly season -- SARIMA can model one of them. That
  limitation is exactly why Fourier seasonal terms, and models like Prophet, exist.}
\end{frame}

\begin{frame}{Choosing the orders -- on our own series}
  \begin{center}
    \includegraphics[width=0.97\textwidth]{fig/acf_pacf_series.pdf}
  \end{center}
  \vskip2pt
  \begin{columns}[T]
    \column{0.52\textwidth}
    \footnotesize
    \textbf{Left:} the raw ACF decays slowly (lag 1 $=0.94$) -- classic ``still
    trending'', so we need $d \ge 1$.
    \vskip3pt
    \textbf{Middle:} after $(1-B)(1-B^{12})$ -- one ordinary and one seasonal
    difference -- one clear \emph{negative} spike at \textbf{lag 12} ($-0.35$).
    A single negative seasonal spike in the ACF is the signature of a
    \textbf{seasonal MA} term, so $Q=1$.
    \column{0.48\textwidth}
    \callout{armblue}{\footnotesize That gives
    $\text{SARIMA}(0,1,1)(0,1,1)_{12}$ -- known as the \textbf{airline model}, because
    Box and Jenkins fitted exactly this to monthly airline passengers.}
  \end{columns}
\end{frame}

\begin{frame}{Did reading the plot actually help?}
  \begin{columns}[T]
    \column{0.46\textwidth}
    We could have guessed a seasonal \emph{AR} term instead. Compare the two by
    \textbf{Akaike Information Criterion} (\textbf{AIC}) -- fit quality penalised
    by the number of parameters, lower is better:
    \vskip6pt
    {\small
    \begin{tabular}{lcc}
      \textbf{orders} & \textbf{AIC} & \textbf{test MAE} \\ \hline
      $(1,1,1)(1,1,0)_{12}$ & 356.4 & 7.24 \\
      $(0,1,1)(0,1,1)_{12}$ & \textbf{343.8} & \textbf{6.07} \\
    \end{tabular}}
    \vskip6pt
    \footnotesize The plot was right: the seasonal MA wins on both, and we never
    touched the test set to find that out.
    \column{0.54\textwidth}
    \includegraphics[width=\textwidth]{fig/arima_forecast.pdf}
    \vskip2pt
    \callout{paramgreen}{\scriptsize The forecast comes with a \textbf{prediction
    interval} -- uncertainty grows with the horizon. A point forecast without an
    interval is only half an answer.}
  \end{columns}
  \vskip2pt
  \footnotesize \textbf{In practice:} don't hand-tune $p,d,q$ at all. Use
  \texttt{pmdarima.auto\_arima} or \texttt{statsforecast.AutoARIMA}, which search
  orders by AIC for you. Read the plot to \emph{understand}, let the search
  \emph{choose}.
\end{frame}

\section{Exponential smoothing}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Exponential smoothing}}\\[4pt]
    \small Weighted averages where the recent past counts most.
  \end{center}
\end{frame}

\begin{frame}{From a simple average to Holt-Winters}
  \begin{columns}[T]
    \column{0.46\textwidth}
    The family is written \textbf{ETS} -- \textbf{Error, Trend, Seasonality}, the three
    components you switch on or off.
    \vskip4pt
    \textbf{Simple Exponential Smoothing} (\textbf{SES}) -- level only. Weights decay
    geometrically into the past:
    \[ \hat{y}_{t+1} = \alpha y_t + (1-\alpha)\hat{y}_t \]
    Flat forecast -- no good if there is a trend or season.
    \vskip4pt
    \textbf{Holt} -- adds a trend term (level $+$ slope). Forecast is a line.
    \vskip4pt
    \textbf{Holt-Winters} -- adds a seasonal term. Level $+$ trend $+$ season,
    each with its own smoothing rate ($\alpha,\beta,\gamma$).
    \column{0.54\textwidth}
    \includegraphics[width=\textwidth]{fig/exp_smoothing.pdf}
  \end{columns}
  \vskip2pt
  \callout{armblue}{\footnotesize ETS and ARIMA overlap but are not the same: ETS is
  built from interpretable \emph{components}, ARIMA from \emph{autocorrelation}. On
  many business series they perform similarly -- try both.}
\end{frame}

\section{Baselines and evaluation}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Baselines and evaluation}}\\[4pt]
    \small A forecast is only ``good'' relative to the dumbest thing that works.
  \end{center}
\end{frame}

\begin{frame}{Always start with a naive baseline}
  \begin{columns}[T]
    \column{0.5\textwidth}
    Before any model, write down the trivial forecast:
    \begin{itemize}\small
      \item \textbf{Naive:} $\hat{y}_{t} = y_{t-1}$ (tomorrow $=$ today).
      \item \textbf{Seasonal naive:} $\hat{y}_{t} = y_{t-m}$
        (this December $=$ last December).
      \item \textbf{Drift:} last value $+$ average change.
    \end{itemize}
    \vskip6pt
    \callout{armred}{\footnotesize If your fancy model can't beat seasonal naive,
    it is not helping. This happens far more often than people admit.}
    \column{0.5\textwidth}
    \textbf{Error metrics}
    \begin{itemize}\small
      \item \textbf{Mean Absolute Error} (\textbf{MAE}) -- in the target's units.
      \item \textbf{Root Mean Squared Error} (\textbf{RMSE}) -- penalizes big misses more.
      \item \textbf{Mean Absolute Percentage Error} (\textbf{MAPE}) -- breaks when
        $y\approx 0$, and punishes over-forecasting more than under-forecasting.
      \item \textbf{Mean Absolute Scaled Error} (\textbf{MASE}) -- scale-free; the one
        to report.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{MASE, by hand}
  \[ \text{MASE} \;=\;
     \frac{\text{MAE of your forecast on the \textbf{test} set}}
          {\text{MAE of the seasonal naive forecast \textbf{in-sample}, on the training set}} \]
  \vskip2pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Denominator} -- walk the \emph{training} series and predict each month with the
    same month last year. Say those errors are $|{-}12|, 20, |{-}18|, 22$:
    \[ \tfrac{12+20+18+22}{4} = 18 \]
    \textbf{Numerator} -- your model's test errors, say $5, |{-}9|, 7, 3$:
    \[ \tfrac{5+9+7+3}{4} = 6 \]
    \column{0.5\textwidth}
    \[ \text{MASE} = \frac{6}{18} = \mathbf{0.33} \]
    Your errors are a third of what naive repetition would cost. \textbf{Below 1 is a
    win, above 1 means you lost to copying last year.}
    \vskip4pt
    \callout{armorange}{\scriptsize Note the asymmetry: the denominator is measured
    \emph{in-sample}, on train. So the seasonal naive evaluated on the \emph{test} set
    does \textbf{not} have to score exactly $1.00$ -- it can land either side.
    Next lecture ours comes out at $1.22$.}
  \end{columns}
\end{frame}

\begin{frame}{Evaluate on the future, never by random split}
  You cannot shuffle-and-split a time series: that trains on the future to predict
  the past.
  \vskip3pt
  \begin{center}
    \includegraphics[width=0.66\textwidth]{fig/time_aware_split.pdf}
  \end{center}
  \vskip1pt
  \callout{armorange}{\footnotesize This ``time-aware validation'' is the hinge between
  the two lectures. We build the full machinery -- leakage, lag features,
  \texttt{TimeSeriesSplit} -- in lecture 31.}
\end{frame}

\begin{frame}{One more trap: even the baseline has to respect the arrow}
  Seasonal naive means ``this December $=$ last December''. Writing that as
  \texttt{y.shift(12)} across a test block looks harmless -- and is wrong the moment
  your horizon exceeds one season.
  \vfill
  \begin{columns}[T]
    \column{0.52\textwidth}
    \callout{armred}{\footnotesize Forecasting 18 months from one origin, steps 13--18
    would reach back only 12 and land \emph{inside the test window} -- the ``baseline''
    reads the future it is supposed to predict.}
    \vskip4pt
    \footnotesize The honest version always reaches back to an \emph{observed} value:
    \[ \hat{y}_{T+k} \;=\; y_{T+k-m\lceil k/m\rceil} \]
    \column{0.48\textwidth}
    \footnotesize On our series that single fix moves the seasonal naive from
    MASE \textbf{0.90} to \textbf{1.22}.
    \vskip4pt
    \callout{paramgreen}{\footnotesize Which is the whole point: a leaky baseline is
    the most dangerous kind. It sets the bar \emph{too high}, so a good model looks
    mediocre -- and you go tune something that was never the problem.}
  \end{columns}
  \vfill
  \footnotesize \textbf{Rule of thumb for picking an approach:} short, single,
  strongly-seasonal series $\to$ ARIMA / ETS are hard to beat. Many related
  series, long history, exogenous drivers $\to$ the ML and deep methods of the
  next lecture start to win.
\end{frame}

\section{Recap}

\begin{frame}{Recap -- classical time series}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Ideas that transfer:}
    \begin{itemize}\small
      \item Time order is information; only train on the past.
      \item Decompose into trend $+$ season $+$ residual (STL).
      \item Difference the \emph{minimum} amount; check with ADF \textbf{and} KPSS.
      \item ACF/PACF read off AR/MA orders; confirm with AIC.
      \item MA($q$) is about \emph{shocks}, not a rolling mean.
      \item ARIMA: autocorrelation-based. ETS: component-based.
      \item Score with MASE -- and make sure the baseline doesn't peek.
    \end{itemize}
    \column{0.5\textwidth}
    \callout{paramgreen}{\textbf{Next (31): ML for time series.}
    Reframe forecasting as supervised learning --
    \begin{itemize}\scriptsize
      \item time-aware splits and leakage,
      \item lag / rolling / calendar features,
      \item \texttt{TimeSeriesSplit} cross-validation,
      \item gradient boosting (and the trend-extrapolation trap),
      \item deep learning and the new foundation models.
    \end{itemize}}
  \end{columns}
\end{frame}
