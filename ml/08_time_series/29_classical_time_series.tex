% Cold open
\begin{frame}{Every model so far assumed the rows were interchangeable}
  Regression, trees, logistic regression -- all of them treat the dataset as a
  \emph{bag} of i.i.d. rows. Shuffle the rows, nothing changes.
  \vfill
  \callout{armred}{\textbf{Time series breaks that assumption.}
  Row order \emph{is} the signal. Yesterday's value tells you a lot about
  today's. You can only train on the past, and you must predict the future --
  never the other way around.}
  \vfill
  Examples everywhere: daily sales, electricity demand, website traffic, the
  dram-to-dollar exchange rate, sensor readings, COVID case counts, your heart rate.
  \vfill
  \footnotesize Two lectures. \textbf{Today (29):} the classical statistical
  toolkit -- decomposition, stationarity, ARIMA, exponential smoothing.
  \textbf{Next (30):} treating forecasting as a supervised ML problem.
\end{frame}

\section{What makes time series different}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{What makes time series different}}\\[4pt]
    \small One column, indexed by time -- and the index carries information.
  \end{center}
\end{frame}

\begin{frame}{The vocabulary}
  A time series is a sequence $y_1, y_2, \dots, y_T$ sampled at regular steps
  (hourly, daily, monthly, \dots).
  \vskip4pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Two jobs:}
    \begin{itemize}\small
      \item \textbf{Forecasting} -- predict future values $y_{T+1}, \dots, y_{T+h}$.
        ($h$ = forecast \emph{horizon}.)
      \item \textbf{Understanding} -- decompose, explain, find structure.
    \end{itemize}
    \vskip4pt
    \textbf{Two shapes of data, and two kinds of input:}
    \begin{itemize}\small
      \item \textbf{Univariate} -- one series, explained by its own past: the
        \textbf{endogenous} input.
      \item \textbf{Multivariate} -- several series, or outside drivers:
        \textbf{exogenous} input.
    \end{itemize}
    \column{0.5\textwidth}
    \callout{armblue}{\scriptsize
      \textbf{The one rule that governs everything:}\\
      information flows forward in time only. Any value, feature, or statistic you
      feed the model must have been \emph{knowable at prediction time}. Violate this
      and you get ``leakage'' -- great backtests, useless in production.}
  \end{columns}
  \vskip2pt
  \callout{armorange}{\scriptsize \textbf{Exogenous} = ``generated outside'':
  temperature for electricity demand, price for sales, a holiday calendar for traffic.
  Everything in lecture 29 is \emph{endogenous}; exogenous drivers arrive in 31.
  \textbf{The catch:} to use one you need its \emph{future} value too, and a
  temperature forecast is itself a forecast.}
\end{frame}

\begin{frame}{The four components of a series}
  \begin{columns}[T]
    \column{0.44\textwidth}
    A series is usefully thought of as a sum (or product) of:
    \begin{itemize}\small
      \item \textbf{Trend} $T_t$ -- long-run drift up or down.
      \item \textbf{Seasonality} $S_t$ -- a pattern that repeats on a \emph{fixed}
        period (24h, 7d, 12mo).
      \item \textbf{Cycle} -- swings with \emph{no} fixed period (business cycles,
        2--10 years). \textbf{Not in the figure}, and not modelled separately: there
        is no calendar to hang it on, so it gets absorbed into $T_t$ or $R_t$.
        Next slide.
      \item \textbf{Noise / residual} $R_t$ -- what's left.
    \end{itemize}
    \vskip4pt
    \footnotesize Additive: $y_t = T_t + S_t + R_t$. \\
    Multiplicative: $y_t = T_t \cdot S_t \cdot R_t$ (season grows with level --
    take $\log$ to make it additive).
    \column{0.56\textwidth}
    \includegraphics[width=\textwidth]{fig/ts_anatomy.pdf}
    \vskip2pt
    {\scriptsize The figure shows the three components this series actually has.}
  \end{columns}
\end{frame}

\begin{frame}{Season or cycle? Ask whether you can put it on a calendar}
  \begin{center}
    \includegraphics[width=0.80\textwidth]{fig/season_vs_cycle.pdf}
  \end{center}
  \vskip2pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \scriptsize
    \textbf{Left -- seasonality.} Fixed, \emph{known} period. Next December is 12
    months after the last one, and you knew that before seeing any data.
    \vskip3pt
    \textbf{Right -- a cycle.} The gaps between peaks are $56, 36, 56, 39$ months.
    Obvious in hindsight, impossible to schedule.
    \column{0.5\textwidth}
    \callout{armorange}{\scriptsize Every seasonal tool here needs a \emph{number}
    $m$ -- seasonal differencing, SARIMA's $m$, Holt-Winters' seasonal term. A cycle
    has no $m$ to give them. \textbf{The trap:} fitting $m=56$ to the right-hand
    panel would confidently schedule a peak that never arrives.}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Decomposition: STL}
  \begin{columns}[T]
    \column{0.44\textwidth}
    \textbf{STL} = ``Seasonal-Trend decomposition using Loess''. It splits the
    series into trend + seasonal + residual, robustly.
    \vskip3pt
    {\scriptsize \textbf{Loess} = \emph{locally estimated scatterplot smoothing}:
    slide a window along the data, fit a tiny regression inside each window, keep only
    its value at the centre, and join those into a curve. No global formula, so it
    bends to whatever shape the data has.}
    \vskip3pt
    {\scriptsize \textbf{Why bother:} see the structure before modelling,
    deseasonalize to read the trend cleanly, and the residual is where anomalies show
    up.}
    \vskip3pt
    {\lstset{basicstyle=\ttfamily\tiny}\begin{lstlisting}
from statsmodels.tsa.seasonal import STL
res = STL(y, period=12).fit()
res.trend, res.seasonal, res.resid
\end{lstlisting}}
    \column{0.56\textwidth}
    \includegraphics[width=\textwidth]{fig/stl_decomposition.pdf}
  \end{columns}
\end{frame}

\section{Stationarity and differencing}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Stationarity and differencing}}\\[4pt]
    \small The classical models need a series whose behaviour does not drift.
  \end{center}
\end{frame}

\begin{frame}{What is stationarity?}
  \begin{columns}[T]
    \column{0.44\textwidth}
    A series is (weakly) \textbf{stationary} if its statistical behaviour does not
    depend on \emph{when} you look:
    \begin{itemize}\small
      \item constant mean (no trend),
      \item constant variance (no growing swings),
      \item the link between two points depends only on \emph{how far apart} they
        are, never on \emph{when} they are.
    \end{itemize}
    \vskip4pt
    \vskip2pt
    {\scriptsize That third condition has a name: the \textbf{autocorrelation} at each
    lag must not drift over time. We measure it two sections from now.}
    \column{0.56\textwidth}
    \includegraphics[width=\textwidth]{fig/stationarity.pdf}
    \vskip2pt
    \callout{armorange}{\scriptsize Why we care: ARMA-type models assume
    stationarity, and for a good reason -- if the relationships keep changing, a
    model fitted on the first half is describing a series that no longer exists in
    the second. Trends and seasonality come out \emph{first}, then get added back
    to the forecast.}
  \end{columns}
\end{frame}

\begin{frame}{Differencing makes a series stationary}
  \begin{columns}[T]
    \column{0.42\textwidth}
    \textbf{Differencing:} model the change instead of the level.
    \[ y'_t = y_t - y_{t-1} \]
    \begin{itemize}\small
      \item Removes a trend.
      \item \textbf{Seasonal differencing} $y_t - y_{t-m}$ removes a season of
        period $m$.
    \end{itemize}
    \vskip4pt
    \textbf{Testing it -- two tests, opposite nulls} {\small(small $p$ rejects, so
    they point opposite ways):}
    \begin{itemize}\scriptsize
      \item \textbf{Augmented Dickey-Fuller} (\textbf{ADF}) -- $H_0$: non-stationary.
      \item \textbf{Kwiatkowski-Phillips-Schmidt-Shin} (\textbf{KPSS}) -- $H_0$:
        stationary.
    \end{itemize}
    \column{0.58\textwidth}
    \includegraphics[width=\textwidth]{fig/differencing.pdf}
  \end{columns}
\end{frame}

\begin{frame}{What does ``difference $d$ times'' actually mean?}
  $d$ is a \textbf{count}: how many times you apply the difference operator, one
  after another. Nothing more mysterious than that.
  \vskip3pt
  \begin{columns}[T]
    \column{0.55\textwidth}
    \small
    \textbf{Why once is normally enough.} Let the trend be a straight line,
    $y_t = a + b\,t$:
    \[ y_t - y_{t-1} = (a + bt) - \big(a + b(t-1)\big) = b \]
    The line is gone in a single pass, and a \emph{constant} is left.
    \vskip3pt
    \textbf{When you need $d=2$.} If the slope itself grows,
    $y_t = a + bt + ct^2$, one difference leaves $b + c(2t-1)$ -- still a line in
    $t$. Difference again and you get $2c$. Hence $d=2$.
    \column{0.45\textwidth}
    \callout{armblue}{\scriptsize In practice $d \in \{0, 1, 2\}$. Wanting $d=3$
    almost never means a trend -- it means a variance that grows (take $\log$) or a
    one-off level shift (model the break).}
    \vskip2pt
    {\scriptsize Seasonal differencing is counted \emph{separately}, as $D$ at
    lag $m$. One of each, $(1-B)(1-B^{12})$, is the usual monthly combination.}
  \end{columns}
  \vskip2pt
  \callout{armred}{\scriptsize \textbf{Stop as soon as the test passes.} On our series
  the standard deviation goes $40.9 \to 10.8 \to \mathbf{11.7}$. The second difference
  had no trend left to remove, so it differenced the \emph{noise} -- and the
  difference of noise is noisier still. Over-differencing costs accuracy and bolts on
  a spurious MA term.}
\end{frame}

\section{ARIMA family}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{The ARIMA family}}\\[4pt]
    \small The workhorse of classical forecasting.
  \end{center}
\end{frame}

\begin{frame}{AR($p$): regress the series on its own past}
  \[ y_t \;=\; c \;+\; \phi_1 y_{t-1} \;+\; \phi_2 y_{t-2} \;+\; \dots
     \;+\; \phi_p y_{t-p} \;+\; \eps_t \]
  \begin{columns}[T]
    \column{0.54\textwidth}
    AR = \textbf{autoregressive}. \textbf{Every symbol, named:}
    \begin{itemize}\small
      \item $y_{t-i}$ -- the value $i$ steps ago. \emph{Observed}: it is sitting in
        your data.
      \item $\phi_i$ (\textbf{phi}) -- the \textbf{weight} on that past value. This
        is what gets fitted. $\phi_1 = 0.8$ reads as ``today starts at $80\%$ of
        yesterday''.
      \item $c$ -- a constant, setting the level the series returns to.
      \item $\eps_t$ (\textbf{epsilon}) -- the \textbf{shock} at time $t$: the part
        of today the past could not predict. Mean $0$, constant variance,
        uncorrelated. All the randomness enters here.
    \end{itemize}
    \column{0.46\textwidth}
    \callout{armblue}{\scriptsize \textbf{One step, by hand.} AR(1) with $c=2$,
    $\phi_1=0.8$. Yesterday $y_{t-1}=10$ and today's shock $\eps_t=-1$ give
    \[ y_t = 2 + 0.8(10) - 1 = \mathbf{9} \]
    \vskip-4pt
    Then with $y_t = 9$ and a shock of $+3$:
    $y_{t+1} = 2 + 0.8(9) + 3 = \mathbf{12.2}$.}
    \vskip2pt
    {\scriptsize \textbf{$\eps$ is not something the fitting picks:} only $c$ and the
    $\phi$'s are chosen, and $\eps_t$ is whatever is left over. Next slide: how.}
    \vskip2pt
    \scriptsize \textbf{What $\phi$'s size means:} $|\phi| < 1$ pulls back toward the
    mean (stationary); $\phi = 1$ is a random walk, shocks \emph{never} fade;
    $|\phi| > 1$ explodes.
  \end{columns}
\end{frame}

\begin{frame}{AR: it really is just least squares}
  \[ y_t = c + \phi_1 y_{t-1} + \phi_2 y_{t-2} + \eps_t \]
  \begin{columns}[T]
    \column{0.5\textwidth}
    Everything on the right is \textbf{in your data}. So lay it out as a table:
    \vskip3pt
    {\small
    \begin{tabular}{cc|c}
      $y_{t-2}$ & $y_{t-1}$ & $y_t$ \\ \hline
      112 & 118 & 132 \\
      118 & 132 & 129 \\
      132 & 129 & 121 \\
      \multicolumn{3}{c}{$\vdots$}
    \end{tabular}}
    \vskip4pt
    That is a design matrix, and this is \emph{exactly} the reframing lecture 30
    opens with. Then it is \textbf{ordinary least squares} (\textbf{OLS}): pick
    $\phi$ to minimise $\sum_t (y_t - \hat{y}_t)^2$, closed form, no search:
    \[ \hat{\bm{\phi}} = (\Xmat^\top\Xmat)^{-1}\Xmat^\top \yv \]
    \column{0.5\textwidth}
    \callout{paramgreen}{\footnotesize \textbf{Measured on our series.} Plain OLS on
    one lag column gives $\phi = \mathbf{0.9686}$. \texttt{statsmodels}' exact
    maximum likelihood gives $\mathbf{0.9749}$. A gap of $0.006$.}
    \vskip3pt
    \footnotesize \textbf{Why any gap at all?} OLS quietly throws away the first
    observations -- it can only start once it has $p$ lags in hand. Exact likelihood
    models those startup values too. With $T = 96$ this barely matters; with
    $T = 20$ it can.
    \vskip3pt
    \callout{armblue}{\scriptsize A second closed-form route, \textbf{Yule-Walker},
    solves for $\phi$ straight from the autocorrelations you plotted earlier. The
    correlogram is not only a diagnostic -- it can hand you the coefficients.}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Interlude: the moving average you already know}
  \begin{center}
    \includegraphics[width=0.80\textwidth]{fig/rolling_mean.pdf}
  \end{center}
  \vskip2pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \scriptsize Replace each point by the mean of the last $k$ values:
    \[ \mathrm{MA}_3(t) = \tfrac{1}{3}\big(y_t + y_{t-1} + y_{t-2}\big) \]
    The window slides one step at a time: $145.0$, then $142.2$, then $141.0$.
    Bigger $k$ smooths harder -- $k=12$ on monthly data erases the season outright,
    which is roughly how a trend estimate gets made.
    \column{0.5\textwidth}
    \callout{armblue}{\scriptsize A \textbf{descriptive smoother}, not a model: no
    parameters to fit, and it makes no forecast by itself.}
    \vskip2pt
    \callout{armorange}{\scriptsize Note for lecture 30:
    \texttt{y.rolling(3).mean()} includes $y_t$ itself, so as a \emph{feature} for
    predicting $y_t$ it is leakage. Shift first:
    \texttt{y.shift(1).rolling(3).mean()}.}
  \end{columns}
\end{frame}

\begin{frame}{MA($q$): regress on your own past \emph{mistakes}}
  \[ y_t \;=\; c \;+\; \eps_t \;+\; \theta_1 \eps_{t-1} \;+\; \dots
     \;+\; \theta_q \eps_{t-q} \]
  \begin{columns}[T]
    \column{0.5\textwidth}
    MA = \textbf{moving average}, a historical misnomer (next slide).
    \begin{itemize}\small
      \item $\eps_t$ -- the same shock as in AR. \textbf{Unobserved}: it is the
        model's own error, so it does not exist until there is a model.
      \item $\theta_j$ (\textbf{theta}) -- how much of the surprise from $j$ steps
        ago is \emph{still echoing} today.
    \end{itemize}
    \vskip3pt
    \callout{armblue}{\footnotesize \textbf{The one-line contrast.} AR carries
    forward the \textbf{value}; MA carries forward the \textbf{surprise}.}
    \column{0.5\textwidth}
    \textbf{Unrolled, by hand.} MA(1), $c=50$, $\theta_1 = 0.7$, with shocks
    $\eps = +4, -2, +6, 0$:
    {\footnotesize
    \begin{align*}
      y_1 &= 50 + 4 &&= 54 \\
      y_2 &= 50 + (-2) + 0.7(4)   &&= 50.8 \\
      y_3 &= 50 + 6 + 0.7(-2)     &&= 54.6 \\
      y_4 &= 50 + 0 + 0.7(6)      &&= 54.2
    \end{align*}}
  \end{columns}
  \vskip1pt
  \callout{armorange}{\footnotesize The big surprise at $t=3$ still lifts $t=4$, and
  is then gone \emph{completely} at $t=5$. An MA($q$) has a memory exactly $q$ steps
  long -- a hard cut-off. An AR's memory decays forever without ever quite reaching
  zero. That difference is what the ACF and PACF are about to expose.}
\end{frame}

\begin{frame}{MA: the same trick breaks, and the way around it}
  \vskip-4pt
  \[ y_t = c + \eps_t + \theta_1 \eps_{t-1} \]
  \vskip-4pt
  \small
  \begin{columns}[T]
    \column{0.48\textwidth}
    Try to build the same table. The first column would have to be $\eps_{t-1}$ --
    and \textbf{it is not in your data}. The shocks are the model's own errors, so
    they do not exist until you already have a model. Circular.
    \vskip3pt
    \callout{paramgreen}{\scriptsize \textbf{The escape, and it is the whole idea:}
    \emph{fix a candidate $\theta_1$ and the shocks become computable.} Set
    $\eps_0 = 0$, then walk forward:
    \[ \eps_t = y_t - c - \theta_1 \eps_{t-1} \]
    \vskip-4pt
    Each step only needs the $\eps$ you just worked out.}
    \column{0.52\textwidth}
    So \emph{any} $\theta_1$ produces a full residual series, which you can score by
    its \textbf{sum of squared errors} (\textbf{SSE}):
    \vskip-6pt
    \[ \mathrm{SSE}(\theta_1) = \sum_t \eps_t^2 \]
    \vskip-4pt
    Hand that function to a numerical optimiser and ask for the minimum.
    \vskip3pt
    This is \textbf{conditional maximum likelihood}: when the shocks are Gaussian,
    maximising the likelihood \emph{is} minimising that sum of squares.
    \vskip2pt
    \callout{armorange}{\scriptsize No formula means a \emph{search}, and searches
    can fail: sensitivity to the starting guess, local minima, non-convergence. That
    is why \texttt{.fit()} sometimes warns, and why the warning is not decoration.}
  \end{columns}
\end{frame}

\begin{frame}{Can we really recover the shocks? And where $|\theta| < 1$ comes from}
  \begin{center}
    \includegraphics[width=0.74\textwidth]{fig/ma_invertibility.pdf}
  \end{center}
  \vskip2pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \scriptsize We seeded that recursion with $\eps_0 = 0$, which is simply
    \emph{false}. It stops mattering, because each step multiplies the leftover error
    by $\theta$: at $\theta = 0.7$ the recovery error falls from $2.2\cdot 10^{-2}$ to
    $1.4\cdot 10^{-8}$ by $t = 40$. Closer to $1$ is slower -- $\theta = 0.9$ is still
    at $4\cdot 10^{-4}$.
    \column{0.5\textwidth}
    \callout{armred}{\scriptsize \textbf{Now push $\theta$ past 1.} The same step
    \emph{multiplies} the error instead: even with the \textbf{correct} $\theta = 1.3$
    the shocks diverge past $10^{5}$. \textbf{Invertibility} ($|\theta| < 1$) is
    precisely what makes the shocks recoverable -- hence the constraint.}
  \end{columns}
  \vskip2pt
  \callout{armblue}{\scriptsize \textbf{And what is \emph{not} estimated:} an MA(1)
  fit searches \textbf{three} numbers -- $c$, $\theta$, $\sigma^2$ -- however long the
  series. The $\eps$ are \textbf{residuals}, not parameters, and they are exactly the
  one-step-ahead forecast errors -- so ``minimise $\sum_t \eps_t^2$'' and ``minimise
  the one-step error'' are one sentence.}
\end{frame}

\begin{frame}{Careful: ``moving average'' means two different things}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \callout{armred}{\textbf{The MA in ARIMA is \emph{not} the rolling mean from
    two slides ago.}}
    \vskip6pt
    \textbf{The one you already know} -- a \emph{rolling} or \emph{moving} average: a
    sliding mean of the \textbf{values}, used to smooth a curve.
    \[ \tfrac{1}{3}\big(y_{t-1}+y_{t-2}+y_{t-3}\big) \]
    You compute it \emph{from data you can see}.
    \column{0.52\textwidth}
    \textbf{MA($q$) in ARIMA} -- a weighted sum of the past \textbf{shocks}
    (the model's own errors):
    \[ y_t = c + \eps_t + \theta_1\eps_{t-1} + \dots \]
    The $\eps$'s are \emph{unobserved}. You never measure them; the fitting procedure
    infers them. That is why MA orders cannot be read off the data directly the way a
    rolling mean can -- and why they need the ACF.
  \end{columns}
  \vfill
  \callout{armorange}{\footnotesize Two unrelated ideas, one name, purely for historical
  reasons. Next lecture we build rolling-mean \emph{features}
  (\texttt{y.shift(1).rolling(7).mean()}) and feed them to a tree -- that is the
  \emph{other} moving average. Keep them apart.}
\end{frame}

\begin{frame}{Putting them together: ARMA and ARIMA}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{ARMA($p,q$)} -- both terms at once, for a series that is
    \emph{already stationary}:
    \[ y_t = c + \underbrace{\sum_{i=1}^{p}\phi_i y_{t-i}}_{\text{past values}}
             + \eps_t + \underbrace{\sum_{j=1}^{q}\theta_j \eps_{t-j}}_{\text{past shocks}} \]
    \vskip2pt
    \textbf{ARIMA($p,d,q$)} adds the ``I'' = \textbf{Integrated} = difference $d$
    times first, so it handles a trending series directly.
    \column{0.5\textwidth}
    \callout{armblue}{\textbf{Where each number comes from:}
    \begin{itemize}\scriptsize
      \item $p$ -- AR order, read from the \textbf{PACF}
      \item $d$ -- differencing, from ADF / KPSS
      \item $q$ -- MA order, read from the \textbf{ACF}
    \end{itemize}
    Those two plots are the next section.}
    \vskip4pt
    \footnotesize ARIMA(0,1,0) is just the random walk -- the ``tomorrow $=$ today''
    baseline. Every fancier model has to beat it.
  \end{columns}
\end{frame}

\begin{frame}{What AR and MA actually look like}
  \begin{center}
    \includegraphics[width=0.96\textwidth]{fig/ar_vs_ma.pdf}
  \end{center}
  \vskip 4pt
  \callout{armblue}{\footnotesize Same machinery, different memory: an \textbf{AR} term makes
  the series \emph{persist} (today leans on yesterday's \emph{value}); an \textbf{MA} term is
  a short \emph{echo} of the last random \emph{shock}. ARMA blends the two.}
\end{frame}

\begin{frame}{SARIMA: adding seasonality}
  Real series are seasonal, so add a second, seasonal triple:
  \[ \text{SARIMA}(p,d,q)(P,D,Q)_m \]
  \begin{itemize}\small
    \item lower-case $(p,d,q)$ -- the short-term, non-seasonal part;
    \item upper-case $(P,D,Q)$ -- the same idea at the seasonal lag;
    \item $m$ -- periods per season (12 monthly, 7 daily-with-weekly, 24 hourly-with-daily).
  \end{itemize}
  \vfill
  \callout{armorange}{\footnotesize One $m$ only. Hourly electricity demand has a daily
  \emph{and} a weekly \emph{and} a yearly season -- SARIMA can model one of them. That
  limitation is exactly why Fourier seasonal terms, and models like Prophet, exist.}
\end{frame}

\section{Autocorrelation: ACF and PACF}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Autocorrelation}}\\[4pt]
    \small How a series correlates with delayed copies of itself.
  \end{center}
\end{frame}

\begin{frame}{Where autocorrelation comes from}
  \textbf{Autocorrelation is simply the measurement of ``yesterday tells you about
  today''} -- the ordinary correlation you already know, applied to the series
  against a shifted copy of itself.
  \vskip2pt
  \begin{center}
    \includegraphics[width=0.86\textwidth]{fig/lag_scatter.pdf}
  \end{center}
  \vskip2pt
  \begin{columns}[T]
    \column{0.52\textwidth}
    \scriptsize \textbf{The recipe:} copy the series, shift the copy $k$ steps, line
    the two up, scatter one against the other, take Pearson's $r$. That number
    \emph{is} the autocorrelation at lag $k$. Repeat for every $k$, draw one bar per
    lag, and you have drawn the \textbf{autocorrelation function} (\textbf{ACF}).
    Nothing else happens in that plot.
    \column{0.48\textwidth}
    \callout{armblue}{\scriptsize Neighbouring months are nearly identical
    ($+0.96$); six months apart is the \emph{weakest} link ($+0.77$), being half a
    season away; twelve months apart is as tight as one ($+0.97$) -- the season
    announcing itself. \textbf{What it buys you:} which lags belong in the model.}
  \end{columns}
\end{frame}

\begin{frame}{ACF and PACF: why you need both}
  \begin{columns}[T]
    \column{0.44\textwidth}
    \callout{armorange}{\scriptsize \textbf{The ACF double-counts.} Today depends on
    yesterday, and yesterday on the day before. So today correlates with the day
    before \emph{for free}, borrowed through yesterday, even with no direct link. The
    ACF reports that borrowed correlation.}
    \vskip2pt
    {\scriptsize The \textbf{partial autocorrelation function} (\textbf{PACF}) is the
    fix: lag $k$ with lags $1 \dots k{-}1$ stripped out. \textbf{ACF} $=$ direct $+$
    borrowed; \textbf{PACF} $=$ direct only.}
    \vskip2pt
    \callout{paramgreen}{\scriptsize \textbf{How lag 1 gets removed}, at $k=2$:
    predict $y_t$ from $y_{t-1}$ and keep the residual $a_t$; predict $y_{t-2}$ from
    $y_{t-1}$ and keep the residual $b_t$; then
    $\mathrm{PACF}(2) = \mathrm{corr}(a_t, b_t)$.
    \vskip2pt
    Both ends have had everything $y_{t-1}$ explains \textbf{subtracted off}, so what
    is left cannot travel through lag 1.}
    \column{0.56\textwidth}
    \includegraphics[width=\textwidth]{fig/acf_pacf.pdf}
    \vskip2pt
    \callout{armblue}{\footnotesize The classic way to read the orders off the data:
    \begin{itemize}\scriptsize
      \item PACF cuts off after lag $p$ $\Rightarrow$ \textbf{AR($p$)}.
      \item ACF cuts off after lag $q$ $\Rightarrow$ \textbf{MA($q$)}.
      \item Both decay $\Rightarrow$ a mix (ARMA).
    \end{itemize}
    That cut-off is exactly the ``memory exactly $q$ steps long'' from the MA slide.}
    \vskip1pt
    {\scriptsize The shaded band is the ``not significantly different from zero''
    region ($\pm 1.96/\sqrt{T}$).}
  \end{columns}
\end{frame}

\begin{frame}{Choosing the orders -- on our own series}
  \begin{center}
    \includegraphics[width=0.97\textwidth]{fig/acf_pacf_series.pdf}
  \end{center}
  \vskip2pt
  \begin{columns}[T]
    \column{0.52\textwidth}
    \footnotesize
    \textbf{Left:} the raw ACF decays slowly (lag 1 $=0.94$) -- classic ``still
    trending'', so we need $d \ge 1$.
    \vskip3pt
    \textbf{Middle:} after $(1-B)(1-B^{12})$ ($B$ is \textbf{backshift},
    $B\,y_t=y_{t-1}$) -- one ordinary and one seasonal difference -- one clear
    \emph{negative} spike at \textbf{lag 12} ($-0.35$).
    A single negative seasonal spike in the ACF is the signature of a
    \textbf{seasonal MA} term, so $Q=1$.
    \column{0.48\textwidth}
    \callout{armblue}{\footnotesize That gives
    $\text{SARIMA}(0,1,1)(0,1,1)_{12}$ -- known as the \textbf{airline model}, because
    Box and Jenkins fitted exactly this to monthly airline passengers.}
  \end{columns}
\end{frame}

\begin{frame}{Did reading the plot actually help?}
  \begin{columns}[T]
    \column{0.46\textwidth}
    We could have guessed a seasonal \emph{AR} term instead. Compare the two by
    \textbf{Akaike Information Criterion} (\textbf{AIC}) -- fit quality penalised
    by the number of parameters, lower is better:
    \vskip6pt
    {\small
    \begin{tabular}{lcc}
      \textbf{orders} & \textbf{AIC} & \textbf{test MAE} \\ \hline
      $(1,1,1)(1,1,0)_{12}$ & 356.4 & 7.24 \\
      $(0,1,1)(0,1,1)_{12}$ & \textbf{343.8} & \textbf{6.07} \\
    \end{tabular}}
    \vskip6pt
    \footnotesize The plot was right: the seasonal MA wins on both, and we never
    touched the test set to find that out.
    \column{0.54\textwidth}
    \includegraphics[width=\textwidth]{fig/arima_forecast.pdf}
    \vskip2pt
    \callout{paramgreen}{\scriptsize The forecast comes with a \textbf{prediction
    interval} -- uncertainty grows with the horizon. A point forecast without an
    interval is only half an answer.}
  \end{columns}
  \vskip2pt
  \footnotesize \textbf{In practice:} don't hand-tune $p,d,q$ at all. Use
  \texttt{pmdarima.auto\_arima} or \texttt{statsforecast.AutoARIMA}, which search
  orders by AIC for you. Read the plot to \emph{understand}, let the search
  \emph{choose}.
\end{frame}

\section{Exponential smoothing}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Exponential smoothing}}\\[4pt]
    \small Weighted averages where the recent past counts most.
  \end{center}
\end{frame}

\begin{frame}{Why average the past at all -- and why not equally?}
  You want one number summarising ``where the series is right now''. Three ways to
  get it:
  \vskip2pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{enumerate}\small
      \item \textbf{Mean of everything.} Uses all the data -- but a value from five
        years ago counts exactly as much as yesterday. Hopelessly sluggish.
      \item \textbf{Mean of the last $k$} (the rolling mean). Reacts, but it is a
        cliff: the $k$-th oldest point counts \emph{fully}, the $(k{+}1)$-th counts
        \emph{zero}, everything older is thrown away.
      \item \textbf{Weight everything, decaying with age.} Nothing discarded,
        nothing old dominating. This is \textbf{exponential smoothing}.
    \end{enumerate}
    \column{0.5\textwidth}
    \includegraphics[width=\textwidth]{fig/ses_weights.pdf}
  \end{columns}
  \vskip2pt
  \callout{armblue}{\scriptsize The weights run $\alpha$, $\alpha(1-\alpha)$,
  $\alpha(1-\alpha)^2, \dots$ -- geometric, hence ``exponential''. $\alpha=0.8$ gives
  $0.80, 0.16, 0.03$ (almost everything on the last two points); $\alpha=0.2$ gives
  $0.20, 0.16, 0.13, 0.10$ (a long memory). \textbf{$\alpha$ is the single dial
  between ``trust the latest reading'' and ``trust the history''.}}
\end{frame}

\begin{frame}{Simple exponential smoothing, written two ways}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Recursive form}
    \[ \hat{y}_{t+1} = \alpha y_t + (1-\alpha)\hat{y}_t \]
    ``the new forecast is the old forecast, blended with what actually happened.''
    \vskip4pt
    \textbf{Error-correction form} -- the same line, rearranged:
    \[ \hat{y}_{t+1} = \hat{y}_t + \alpha\big(y_t - \hat{y}_t\big) \]
    Read it: $\alpha$ is \emph{the fraction of your latest error you correct for}.
    At $\alpha = 0.2$ you close a fifth of the gap.
    \column{0.5\textwidth}
    \textbf{Where the geometric weights come from.} Substitute the rule into itself:
    {\footnotesize
    \begin{align*}
      \hat{y}_{t+1} &= \alpha y_t + (1-\alpha)\hat{y}_t \\
                    &= \alpha y_t + (1-\alpha)\big[\alpha y_{t-1} + (1-\alpha)\hat{y}_{t-1}\big] \\
                    &= \alpha y_t + \alpha(1-\alpha) y_{t-1} + (1-\alpha)^2 \hat{y}_{t-1}
    \end{align*}}
    \vskip-2pt
    Keep going and you get the weights on the previous slide. The recursion
    \emph{is} the infinite weighted average, computed two numbers at a time.
  \end{columns}
  \vskip2pt
  \callout{armorange}{\footnotesize \textbf{The limits:} $\alpha \to 1$ gives
  $\hat{y}_{t+1} = y_t$, the naive forecast. $\alpha \to 0$ freezes at a constant.
  And beyond one step the forecast is \textbf{flat}: $\hat{y}_{t+h} = \hat{y}_{t+1}$
  for every $h$. That flatness is precisely why Holt and Holt-Winters exist.}
\end{frame}

\begin{frame}{From a simple average to Holt-Winters}
  \begin{columns}[T]
    \column{0.46\textwidth}
    The family is written \textbf{ETS} -- \textbf{Error, Trend, Seasonality}, the three
    components you switch on or off.
    \vskip4pt
    \textbf{Simple Exponential Smoothing} (\textbf{SES}) -- level only, one dial
    $\alpha$. Flat forecast.
    \vskip4pt
    \textbf{Holt} -- adds a \emph{slope}, smoothed by its own dial $\beta$ exactly the
    way $\alpha$ smooths the level. Forecast is a line.
    \vskip4pt
    \textbf{Holt-Winters} -- adds a \emph{seasonal profile}, smoothed by $\gamma$.
    Level $+$ trend $+$ season, three dials, one idea applied three times.
    \column{0.54\textwidth}
    \includegraphics[width=\textwidth]{fig/exp_smoothing.pdf}
  \end{columns}
  \vskip2pt
  \callout{armblue}{\footnotesize ETS and ARIMA overlap but are not the same: ETS is
  built from interpretable \emph{components}, ARIMA from \emph{autocorrelation}. On
  many business series they perform similarly -- try both.}
\end{frame}

\begin{frame}{By hand: choosing $\alpha$ for exponential smoothing}
  Series $y = 10,\, 14,\, 12,\, 18,\, 16$. Start the level at $\hat{y}_1 = y_1 = 10$
  and score with the summed squared \textbf{one-step-ahead} errors.
  \vskip3pt
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{$\alpha = 0.2$} (trust the history)
    \vskip2pt
    {\footnotesize
    \begin{tabular}{c|ccc}
      $t$ & forecast & actual & error \\ \hline
      2 & 10.00 & 14 & $+4.00$ \\
      3 & 10.80 & 12 & $+1.20$ \\
      4 & 11.04 & 18 & $+6.96$ \\
      5 & 12.43 & 16 & $+3.57$ \\ \hline
      \multicolumn{3}{r}{\textbf{SSE}} & \textbf{78.61}
    \end{tabular}}
    \column{0.48\textwidth}
    \textbf{$\alpha = 0.8$} (trust the latest reading)
    \vskip2pt
    {\footnotesize
    \begin{tabular}{c|ccc}
      $t$ & forecast & actual & error \\ \hline
      2 & 10.00 & 14 & $+4.00$ \\
      3 & 13.20 & 12 & $-1.20$ \\
      4 & 12.24 & 18 & $+5.76$ \\
      5 & 16.85 & 16 & $-0.85$ \\ \hline
      \multicolumn{3}{r}{\textbf{SSE}} & \textbf{51.34}
    \end{tabular}}
  \end{columns}
  \vskip4pt
  \callout{armblue}{\footnotesize One update in full:
  $\hat{y}_3 = 0.8(14) + 0.2(10) = 11.2 + 2.0 = \mathbf{13.2}$.
  \quad $\alpha = 0.8$ wins on this series -- it jumps around, so leaning on the
  latest value pays. Notice we did not \emph{choose} that; we \textbf{scored} it.}
\end{frame}

\begin{frame}{Same recipe everywhere}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \includegraphics[width=\textwidth]{fig/alpha_search.pdf}
    \vskip2pt
    \footnotesize The two squares are the tables you just filled in. Sweep every
    $\alpha$ and the minimum sits at $\alpha = \mathbf{0.67}$ (SSE $49.8$). An
    optimiser simply walks downhill to it.
    \column{0.48\textwidth}
    \callout{paramgreen}{\textbf{The recipe, unchanged all lecture:}
    write the one-step-ahead error, square it, sum it, minimise it.}
    \vskip4pt
    \begin{itemize}\small
      \item \textbf{AR} -- regressors observed $\Rightarrow$ \emph{closed form}.
      \item \textbf{MA / ARMA / SARIMA} -- regressors unobserved $\Rightarrow$
        \emph{numerical search}.
      \item \textbf{ETS} ($\alpha,\beta,\gamma$) -- numerical search, and the
        starting level / trend / season get fitted too.
    \end{itemize}
    \vskip3pt
    \callout{armorange}{\scriptsize \textbf{Do not confuse two loops.} Fitting picks
    the \emph{coefficients} for one fixed $(p,d,q)$. \textbf{AIC} picks between
    \emph{different} $(p,d,q)$. Inner loop and outer loop.}
  \end{columns}
  \vskip1pt
  {\scriptsize Real libraries refine the inner loop, but the shape never changes:
  \emph{propose, score, improve.}}
\end{frame}

\section{Baselines and evaluation}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Baselines and evaluation}}\\[4pt]
    \small A forecast is only ``good'' relative to the dumbest thing that works.
  \end{center}
\end{frame}

\begin{frame}{Always start with a naive baseline}
  \begin{columns}[T]
    \column{0.5\textwidth}
    Before any model, write down the trivial forecast:
    \begin{itemize}\small
      \item \textbf{Naive:} $\hat{y}_{t} = y_{t-1}$ (tomorrow $=$ today).
      \item \textbf{Seasonal naive:} $\hat{y}_{t} = y_{t-m}$
        (this December $=$ last December).
      \item \textbf{Drift:} last value $+$ average change.
    \end{itemize}
    \vskip6pt
    \callout{armred}{\footnotesize If your fancy model can't beat seasonal naive,
    it is not helping. This happens far more often than people admit.}
    \column{0.5\textwidth}
    \textbf{Error metrics}
    \begin{itemize}\small
      \item \textbf{Mean Absolute Error} (\textbf{MAE}) -- in the target's units.
      \item \textbf{Root Mean Squared Error} (\textbf{RMSE}) -- penalizes big misses more.
      \item \textbf{Mean Absolute Percentage Error} (\textbf{MAPE}) -- breaks when
        $y\approx 0$, and punishes over-forecasting more than under-forecasting.
      \item \textbf{Mean Absolute Scaled Error} (\textbf{MASE}) -- scale-free; the one
        to report.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{MASE, by hand}
  \[ \text{MASE} \;=\;
     \frac{\text{MAE of your forecast on the \textbf{test} set}}
          {\text{MAE of the seasonal naive forecast \textbf{in-sample}, on the training set}} \]
  \vskip2pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Denominator} -- walk the \emph{training} series and predict each month with the
    same month last year. Say those errors are $|{-}12|, 20, |{-}18|, 22$:
    \[ \tfrac{12+20+18+22}{4} = 18 \]
    \textbf{Numerator} -- your model's test errors, say $5, |{-}9|, 7, 3$:
    \[ \tfrac{5+9+7+3}{4} = 6 \]
    \column{0.5\textwidth}
    \[ \text{MASE} = \frac{6}{18} = \mathbf{0.33} \]
    Your errors are a third of what naive repetition would cost. \textbf{Below 1 is a
    win, above 1 means you lost to copying last year.}
    \vskip4pt
    \callout{armorange}{\scriptsize Note the asymmetry: the denominator is measured
    \emph{in-sample}, on train. So the seasonal naive evaluated on the \emph{test} set
    does \textbf{not} have to score exactly $1.00$ -- it can land either side.
    Two slides from now ours comes out at $1.22$.}
  \end{columns}
\end{frame}

\begin{frame}{Evaluate on the future, never by random split}
  You cannot shuffle-and-split a time series: that trains on the future to predict
  the past.
  \vskip3pt
  \begin{center}
    \includegraphics[width=0.66\textwidth]{fig/time_aware_split.pdf}
  \end{center}
  \vskip1pt
  \callout{armorange}{\scriptsize The \textbf{k-fold cross-validation} you have used
  since Ridge assumes exchangeable rows. Shuffling breaks that \emph{twice}: you train
  on the future, \emph{and} adjacent rows are \textbf{autocorrelated}, so neighbouring
  folds leak into each other. Full machinery -- lag features,
  \texttt{TimeSeriesSplit} -- in lecture 30.}
\end{frame}

\begin{frame}{One more trap: even the baseline has to respect the arrow}
  Seasonal naive means ``this December $=$ last December''. Writing that as
  \texttt{y.shift(12)} across a test block looks harmless -- and is wrong the moment
  your horizon exceeds one season.
  \vfill
  \begin{columns}[T]
    \column{0.52\textwidth}
    \callout{armred}{\footnotesize Forecasting 18 months from one origin, steps 13--18
    would reach back only 12 and land \emph{inside the test window} -- the ``baseline''
    reads the future it is supposed to predict.}
    \vskip4pt
    \footnotesize The honest version always reaches back to an \emph{observed} value:
    \[ \hat{y}_{T+k} \;=\; y_{T+k-m\lceil k/m\rceil} \]
    \column{0.48\textwidth}
    \footnotesize On our series that single fix moves the seasonal naive from
    MASE \textbf{0.90} to \textbf{1.22}.
    \vskip4pt
    \callout{paramgreen}{\footnotesize Which is the whole point: a leaky baseline is
    the most dangerous kind. It sets the bar \emph{too high}, so a good model looks
    mediocre -- and you go tune something that was never the problem.}
  \end{columns}
  \vfill
  \footnotesize \textbf{Rule of thumb for picking an approach:} short, single,
  strongly-seasonal series $\to$ ARIMA / ETS are hard to beat. Many related
  series, long history, exogenous drivers $\to$ the ML and deep methods of the
  next lecture start to win.
\end{frame}

\section{Recap}

\begin{frame}{Recap -- classical time series}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \textbf{Ideas that transfer:}
    \begin{itemize}\small
      \item Time order is information; only train on the past.
      \item Decompose into trend $+$ season $+$ residual (STL). A \emph{cycle} has no
        fixed period, so it never gets its own term.
      \item Difference the \emph{minimum} amount; check with ADF \textbf{and} KPSS.
      \item AR carries the \emph{value} forward, MA the \emph{shock}. MA($q$) is not
        a rolling mean.
      \item Autocorrelation is just $r$ between a series and a shifted copy; ACF is
        the total link, PACF the direct one.
      \item Coefficients are not chosen by taste: \textbf{minimise the summed squared
        one-step error}. AR has a closed form; everything else is a search.
      \item Score with MASE -- and make sure the baseline doesn't peek.
    \end{itemize}
    \column{0.48\textwidth}
    \callout{paramgreen}{\textbf{Next (30): ML for time series.}
    Reframe forecasting as supervised learning --
    \begin{itemize}\scriptsize
      \item time-aware splits and leakage,
      \item lag / rolling / calendar features,
      \item \texttt{TimeSeriesSplit} cross-validation,
      \item gradient boosting (and the trend-extrapolation trap),
      \item deep learning and the new foundation models.
    \end{itemize}}
    \vskip4pt
    \footnotesize The lag table on the ``AR is least squares'' slide is literally the
    first slide of lecture 30. Classical and ML forecasting are closer than the two
    names suggest.
  \end{columns}
\end{frame}
