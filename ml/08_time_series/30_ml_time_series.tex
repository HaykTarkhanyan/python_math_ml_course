\begin{frame}{Where we left off}
  Last time: ARIMA, SARIMA, exponential smoothing -- purpose-built statistical
  models. They are excellent, and often hard to beat on a single clean series.
  \vfill
  \callout{armblue}{\textbf{Today's idea:} we already own a toolbox of powerful
  regressors -- linear models, random forests, gradient boosting, neural nets.
  What if we could point \emph{those} at a time series?}
  \vfill
  We can. The trick is to turn ``predict the next value'' into an ordinary
  $(\Xmat, \yv)$ table. Once it's a table, every model from this course applies.
  \vfill
  \footnotesize The catch: the i.i.d. assumption is still false, so we have to be
  very careful about \emph{how} we build the table and \emph{how} we validate.
  That discipline is most of this lecture.
\end{frame}

\section{Forecasting as supervised learning}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Forecasting as supervised learning}}\\[4pt]
    \small Slide a window over the series and out falls a design matrix.
  \end{center}
\end{frame}

\begin{frame}{The reframing}
  Pick a target ``now'' ($y_t$) and describe it with values from its own past
  ($y_{t-1}, y_{t-2}, \dots$). Each time step becomes one training row.
  \vfill
  \begin{center}
    \includegraphics[width=0.92\textwidth]{fig/supervised_reframe.pdf}
  \end{center}
  \vfill
  \callout{paramgreen}{\footnotesize That's it -- the columns on the left are
  features, the right column is the label. Feed it to \texttt{LinearRegression},
  \texttt{RandomForest}, \texttt{HistGradientBoosting} -- whatever you like.}
\end{frame}

\section{Leakage and time-aware splitting}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Leakage and time-aware splitting}}\\[4pt]
    \small The single most common way to fool yourself with time series.
  \end{center}
\end{frame}

\begin{frame}{Predict-first: what's wrong with a random split?}
  You have your $(\Xmat, \yv)$ table. Instinct says
  \texttt{train\_test\_split(X, y, shuffle=True)}.
  \vfill
  \textbf{Predict: what does a random split do to a time series?}
  \pause
  \vfill
  \begin{columns}[T]
    \column{0.5\textwidth}
    \callout{armred}{\footnotesize It puts \emph{future} rows in the training set
    and \emph{past} rows in the test set. The model peeks at the future to
    ``predict'' the past. Your test score is a fantasy.}
    \column{0.5\textwidth}
    \footnotesize Worse, the rows aren't even independent: $y_{t}$ and $y_{t-1}$
    share almost everything, so a random test point sits right next to its own
    neighbours in train. Textbook \textbf{data leakage}.
  \end{columns}
  \vfill
  \footnotesize The fix is boring and non-negotiable: \textbf{split by time}.
\end{frame}

\begin{frame}{Random split vs forward split}
  \begin{center}
    \includegraphics[width=0.82\textwidth]{fig/time_aware_split.pdf}
  \end{center}
  \vfill
  \callout{paramgreen}{\footnotesize Train on an earlier block, test on the later
  block. Same rule as lecture 29: only ever learn from the past. This also means
  \textbf{no shuffling}, and any scaler / encoder must be fit on train only.}
\end{frame}

\section{Feature engineering for time}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Feature engineering for time}}\\[4pt]
    \small This is where an ML forecaster is won or lost.
  \end{center}
\end{frame}

\begin{frame}{The four families of time features}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{1. Lags} -- past values.
    \begin{itemize}\scriptsize
      \item \texttt{y.shift(1)}, \texttt{shift(7)}, \texttt{shift(12)}.
      \item Pick lags at meaningful horizons (yesterday, last week, last year).
    \end{itemize}
    \vskip3pt
    \textbf{2. Rolling / window stats} -- local summaries.
    \begin{itemize}\scriptsize
      \item \texttt{y.shift(1).rolling(7).mean()} / std / min / max.
      \item \textbf{Always shift before rolling} -- else the window includes $y_t$
        itself = leakage.
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{3. Calendar / date parts} -- known in advance.
    \begin{itemize}\scriptsize
      \item month, day-of-week, hour, is\_weekend, is\_holiday.
      \item Cyclically encode: $\sin/\cos(2\pi \cdot \text{hour}/24)$.
    \end{itemize}
    \vskip3pt
    \textbf{4. Exogenous regressors} -- outside drivers.
    \begin{itemize}\scriptsize
      \item price, promotion, weather, events.
      \item The big advantage of ML over plain ARIMA.
    \end{itemize}
  \end{columns}
  \vskip4pt
  \callout{armorange}{\scriptsize Golden rule: every feature for row $t$ must be
  computable using \emph{only} information available strictly before $t$.}
\end{frame}

\section{Time-aware cross-validation}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Time-aware cross-validation}}\\[4pt]
    \small K-fold, but respecting the arrow of time.
  \end{center}
\end{frame}

\begin{frame}[fragile]{TimeSeriesSplit}
  Ordinary K-fold shuffles -- forbidden here. Use forward-chaining: each fold
  trains on an expanding past and validates on the next block.
  \vskip1pt
  \begin{center}
    \includegraphics[width=0.86\textwidth]{fig/ts_cv.pdf}
  \end{center}
  \vskip1pt
  \begin{columns}[T]
    \column{0.42\textwidth}
    {\lstset{basicstyle=\ttfamily\scriptsize}\begin{lstlisting}
TimeSeriesSplit(n_splits=5,
                gap=3)
\end{lstlisting}}
    \column{0.58\textwidth}
    \scriptsize
    \textbf{\texttt{max\_train\_size}} -- fix the window instead of growing it, when
    the distant past is stale.
    \vskip2pt
    \textbf{\texttt{gap}} -- drop the rows straddling the boundary. A
    \texttt{rolling(7)} feature at the start of validation was built from the last
    training days, so the two sets \emph{overlap} even after an honest split.
  \end{columns}
\end{frame}

\section{Tree models and the extrapolation trap}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Tree models and the extrapolation trap}}\\[4pt]
    \small Gradient boosting is the tabular champion -- with one sharp edge.
  \end{center}
\end{frame}

\begin{frame}{Predict-first: point gradient boosting at a trending series}
  \textbf{Gradient boosting} -- a \textbf{Gradient Boosting Machine}, \textbf{GBM} on
  the charts that follow -- wins most tabular competitions (XGBoost / LightGBM /
  \texttt{HistGradientBoosting}). We built lag $+$ calendar features. Let it forecast
  the trending series from lecture 29.
  \vfill
  \textbf{Predict: does the boosting model track the upward trend?}
  \pause
  \vfill
  \begin{center}
    \includegraphics[width=0.9\textwidth]{fig/gbm_forecast.pdf}
  \end{center}
\end{frame}

\begin{frame}{Trees cannot extrapolate -- so difference first}
  \begin{center}
    \includegraphics[width=0.68\textwidth]{fig/gbm_forecast.pdf}
  \end{center}
  \vskip1pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \scriptsize A tree predicts by \emph{averaging training labels} in a leaf, so its
    output can never exceed the largest $y$ it ever saw -- and it settles well
    \emph{below} that ceiling, at the mean of its top leaf ($\approx 200$, against a
    training max of $220$).
    \vskip3pt
    Same lesson as ARIMA's ``I''. Linear models and neural networks extrapolate fine;
    \emph{trees do not}.
    \column{0.5\textwidth}
    \callout{paramgreen}{\scriptsize \textbf{The fix:} make the \emph{target}
    stationary -- predict the difference $y_t - y_{t-1}$, then add it back. Now the
    targets live in a fixed range trees can handle: MAE $19.0 \to 6.7$.}
  \end{columns}
\end{frame}

\begin{frame}{Forecasting more than one step ahead}
  For horizon $h > 1$ you have two strategies:
  \vfill
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Recursive} -- predict $t{+}1$, feed it back in as a lag, predict
    $t{+}2$, \dots
    \begin{itemize}\scriptsize
      \item One model, reused.
      \item Errors compound down the horizon.
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Direct} -- train a separate model per horizon ($h{=}1$ model,
    $h{=}2$ model, \dots).
    \begin{itemize}\scriptsize
      \item No error feedback.
      \item More models to train and store.
    \end{itemize}
  \end{columns}
  \vfill
  \callout{armorange}{\footnotesize You rarely hand-code this. Libraries like
  \texttt{skforecast}, \texttt{mlforecast} (Nixtla), \texttt{sktime} and
  \texttt{Darts} wrap any scikit-learn regressor into a recursive/direct
  forecaster with the leakage-safe plumbing built in.}
\end{frame}

\section{Industry standards}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{What people actually run}}\\[4pt]
    \small The tools you will meet in a real forecasting job.
  \end{center}
\end{frame}

\begin{frame}{Prophet: a curve, not a process}
  \begin{columns}[T]
    \column{0.54\textwidth}
    \includegraphics[width=\textwidth]{fig/prophet_components.pdf}
    \column{0.46\textwidth}
    \small \textbf{Prophet} (Taylor \& Letham, 2018 -- Meta) fits an additive,
    decomposable model:
    \vskip-6pt
    \[ y_t = g(t) + s(t) + h(t) + \eps_t \]
    \vskip-4pt
    \begin{itemize}\small
      \item $g(t)$ -- trend, piecewise linear
      \item $s(t)$ -- seasonality, Fourier terms
      \item $h(t)$ -- holidays, dummy variables
    \end{itemize}
    \vskip2pt
    \callout{armblue}{\scriptsize \textbf{The break with everything so far.} ARIMA and
    ETS model the \emph{process} -- how today depends on yesterday. Prophet regresses
    on \textbf{time itself}. There is not a single lag in it.}
  \end{columns}
  \vskip1pt
  \callout{paramgreen}{\scriptsize That has a practical payoff: gaps, irregular
  spacing and outliers do not break it, because row $t$ never needs row $t-1$.}
\end{frame}

\begin{frame}{The trend is piecewise linear -- and that is the sharp edge}
  \begin{center}
    \includegraphics[width=0.76\textwidth]{fig/prophet_changepoints.pdf}
  \end{center}
  \vskip2pt
  \begin{columns}[T]
    \column{0.52\textwidth}
    \scriptsize Prophet lays down \textbf{25 candidate changepoints} across the first
    \textbf{80\%} of the history, then a sparse prior shrinks most of them to zero so
    only a few survive. How eagerly is set by
    \texttt{changepoint\_prior\_scale} (default $0.05$).
    \column{0.48\textwidth}
    \callout{armred}{\scriptsize \textbf{The forecast is the last segment, extended.}
    Above, the final segment slopes $-0.07$/day, so the forecast keeps falling while
    the series turns back up -- finishing \textbf{37 units} low. A recent blip becomes
    your long-run view.}
  \end{columns}
\end{frame}

\begin{frame}{Fourier seasonality: many seasons at once}
  \begin{center}
    \includegraphics[width=0.84\textwidth]{fig/fourier_seasonality.pdf}
  \end{center}
  \vskip2pt
  \begin{columns}[T]
    \column{0.52\textwidth}
    \scriptsize $s(t)$ is a sum of $\sin/\cos$ pairs. Its \textbf{order} sets how
    sharp a shape it can bend into: on the left, RMSE $4.7 \to 3.5 \to 1.7$ for orders
    $1, 3, 10$. Defaults are order 10 yearly, 3 weekly. Holidays are separate dummies,
    each with its own coefficient.
    \column{0.48\textwidth}
    \callout{paramgreen}{\scriptsize \textbf{This answers a limit lecture 29 left
    open.} SARIMA carries \emph{one} $m$. Fourier terms are just regressors, so daily
    $+$ weekly $+$ yearly can all sit in one model at once.}
  \end{columns}
\end{frame}

\begin{frame}{The honest verdict on Prophet}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{What it is genuinely good at}
    \begin{itemize}\small
      \item Fast and fully automatic -- a sane forecast with no tuning.
      \item Missing data, outliers, irregular spacing.
      \item Multiple seasonalities and holiday calendars.
      \item Components you can \emph{show} a stakeholder.
    \end{itemize}
    \vskip2pt
    {\scriptsize Built at Meta so that analysts who are \emph{not} forecasters could
    produce hundreds of forecasts. That is the problem it solves.}
    \column{0.5\textwidth}
    \callout{armred}{\scriptsize \textbf{What the textbook says.} Hyndman \&
    Athanasopoulos (\emph{fpp3} \S12.2) find Prophet worse than \textbf{both} ETS and
    ARIMA on their example, with enough leftover residual autocorrelation to leave its
    prediction intervals \textbf{too narrow}. Their verdict: it rarely gives better
    forecast accuracy than the alternatives.}
    \vskip2pt
    \callout{armorange}{\scriptsize So: a strong, cheap \textbf{default} -- not a
    winner. Same rule as everything else here, score it against seasonal naive before
    you believe it.}
  \end{columns}
\end{frame}

\begin{frame}{The rest of the production stack}
  \begin{columns}[T]
    \column{0.56\textwidth}
    {\footnotesize
    \begin{tabular}{@{}ll@{}}
      \textbf{Library} & \textbf{Reach for it when} \\ \hline
      \texttt{statsforecast}  & classical models, many series, fast \\
      \texttt{mlforecast}     & sklearn regressor $+$ lag features \\
      \texttt{neuralforecast} & deep models behind one API \\
      \texttt{skforecast}     & recursive/direct around sklearn \\
      \texttt{sktime}         & sklearn-style pipelines, unified API \\
      \texttt{Darts}          & one API over classical $+$ deep \\
      \texttt{GluonTS}        & probabilistic / deep (AWS) \\
      \texttt{Prophet}        & business series, holidays, no tuning \\
    \end{tabular}}
    \column{0.44\textwidth}
    \callout{armblue}{\scriptsize The first three are \textbf{Nixtla}'s stack and
    share one API. Nixtla's own benchmarks put \texttt{statsforecast}'s AutoARIMA
    orders of magnitude ahead of Prophet on speed -- a vendor number, so check it on
    your own series, but the gap is real and it is why the stack spread.}
  \end{columns}
  \vskip2pt
  \callout{paramgreen}{\footnotesize None of these change the rules: time-ordered
  splits, features that were knowable at $t$, and an honest baseline.}
\end{frame}

\section{Deep learning and foundation models}

\begin{frame}[plain]
  \begin{center}
    {\Large\color{popblue}\textbf{Deep learning and foundation models}}\\[4pt]
    \small Awareness level -- what exists and when it's worth it.
  \end{center}
\end{frame}

\begin{frame}{The deep learning lineage}
  \begin{center}
    \includegraphics[width=0.80\textwidth]{fig/deep_ts_timeline.pdf}
  \end{center}
  \vskip1pt
  \begin{columns}[T]
    \column{0.5\textwidth}
    \footnotesize \textbf{Specialist deep models}
    \begin{itemize}\scriptsize
      \item \textbf{DeepAR} -- a \textbf{recurrent} network (it carries a hidden state
        along the sequence), trained across many series, predicting a
        \emph{distribution} not a point.
      \item \textbf{N-BEATS / N-HiTS} -- deep stacks of plain layers.
      \item \textbf{Temporal Fusion Transformer} (\textbf{TFT}) -- attention $+$
        built-in importances.
      \item \textbf{PatchTST} -- patch the series, feed a Transformer.
    \end{itemize}
    \column{0.5\textwidth}
    \footnotesize \textbf{Why they connect to us}
    \vskip2pt
    \scriptsize Recurrence and attention are the subject of two later chapters, so
    treat these as a map, not a syllabus. What matters today: they need a \emph{lot} of
    data -- many series, long history -- to beat a well-tuned gradient booster. On a
    single short series like ours, they usually don't.
  \end{columns}
\end{frame}

\begin{frame}{Foundation models: zero-shot forecasting}
  Since 2023: pre-train one Transformer on billions of time points, then forecast a
  \emph{new} series with no training at all -- the same trick a
  \textbf{Large Language Model} (\textbf{LLM}) does when it answers a question it was
  never specifically trained on.
  \vfill
  \begin{columns}[T]
    \column{0.52\textwidth}
    \footnotesize \textbf{The current crop (mid-2026)}
    \begin{itemize}\scriptsize
      \item \textbf{TimesFM} (Google) -- patched decoder, pre-trained on $\sim$100B points.
      \item \textbf{Chronos-2} (Amazon) -- tokenizes values like language; five sizes
        from 9M to 710M parameters; handles covariates.
      \item \textbf{Moirai-2} (Salesforce) -- multivariate, any frequency.
      \item \textbf{TimeGPT} (Nixtla), \textbf{Sundial}, \textbf{Time-MoE},
        \textbf{Lag-Llama}.
    \end{itemize}
    \vskip2pt
    \scriptsize Don't take a vendor's word for it -- compare on a public leaderboard
    (\textbf{GIFT-Eval}, \textbf{ProbTS}) \emph{and} on your own series.
    \column{0.48\textwidth}
    \callout{armorange}{\scriptsize \textbf{Reality check.} Zero-shot is astonishing
    for a cold start with no history. But a gradient booster with good features and
    exogenous drivers is still a very strong, cheap, interpretable baseline -- and
    today it beat a tuned SARIMA by nothing at all. Try the simple thing first.}
    \vskip4pt
    \callout{armred}{\scriptsize This slide dates fastest of anything in the chapter.
    Check the names before re-lecturing.}
  \end{columns}
\end{frame}

\section{Recap}

\begin{frame}{Recap -- ML for time series}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \begin{itemize}\small
      \item Forecasting $\to$ supervised learning by sliding a window.
      \item \textbf{Never} random-split; split and cross-validate by time
        (\texttt{TimeSeriesSplit}, with a \texttt{gap} if your features roll).
      \item Features: lags, (shifted) rolling stats, calendar, exogenous.
      \item Trees can't extrapolate -- difference the target first.
      \item Multi-step: recursive vs direct (let a library handle it).
      \item Report MASE vs an \emph{honest} baseline (lecture 29's rule still binds).
      \item \textbf{Prophet} regresses on \emph{time} -- piecewise trend $+$ Fourier
        seasonality $+$ holidays. Automatic, multi-seasonal, and rarely the most
        accurate.
      \item Know the stack: Nixtla, \texttt{sktime}, \texttt{Darts}.
    \end{itemize}
    \column{0.48\textwidth}
    \callout{paramgreen}{\textbf{The one habit to keep:}
    \vskip3pt
    every feature and every split must respect the arrow of time. If a value
    wasn't knowable yet, it can't be in the row.}
    \vskip6pt
    \footnotesize \textbf{Toolbox:} \texttt{sktime}, \texttt{statsforecast} \&
    \texttt{mlforecast} (Nixtla), \texttt{skforecast}, \texttt{Darts},
    \texttt{Prophet}, \texttt{GluonTS}.
  \end{columns}
\end{frame}
