\begin{frame}{How do you \emph{look} at 784-dimensional data?}
  \begin{columns}[T]
    \column{0.50\textwidth}
    A photo of a shirt is $28\times28 = \mathbf{784}$ numbers; gene expression $\sim$20{,}000;
    a text embedding $\sim$768. You cannot plot that.
    \vskip 6pt
    \textbf{Dimensionality reduction (DR):} replace many features with a \emph{few} new ones that
    keep most of the structure.
    \vskip 6pt
    Bridge from clustering --- both are \textbf{unsupervised} (no $y$): clustering groups the
    \textbf{rows} (samples), DR compresses the \textbf{columns} (features).
    \column{0.52\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig/dr_fashion_samples.pdf}\\[3pt]
      {\scriptsize \textbf{Running dataset:} Fashion-MNIST --- 12{,}000 of its 70{,}000 garment
      photos ($28\times28$ grayscale, 10 classes). Each image $=$ 784 pixel values; we will
      squeeze those 784 down to 2 to \emph{see} the data.}
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}{Why reduce dimensions?}
  \begin{itemize}
    \item \textbf{Visualization} --- see high-dimensional data in 2-D.
    \item \textbf{Preprocessing / compression} --- faster models, less storage, fewer features
      before clustering or a classifier.
    \item \textbf{Denoising} --- drop tiny-variance directions that are \emph{often} just noise.
  \end{itemize}
  \vfill
  \footnotesize And a deeper reason, next: in high dimensions, distance itself stops working.
\end{frame}

\begin{frame}{The curse of dimensionality}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \textbf{Predict-first:} in 100 dimensions, how much closer is your \emph{nearest} point than
    your \emph{farthest}?
    \pause
    \vskip 4pt
    Almost the same. In 100 dimensions the farthest point is only about \textbf{40\% farther}
    than the nearest; in 2 dimensions it is roughly \textbf{87 times} farther. Distances
    \textbf{concentrate}, so distance-based methods (k-means, kNN, DBSCAN\dots) degrade.
    \vskip 4pt
    \footnotesize Reduce first, and distances mean something again.
    \vskip 3pt
    {\scriptsize Full treatment in our homework: \texttt{math/30\_curse\_of\_dimensionality.qmd}.}
    \column{0.48\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig/dr_curse_distances.pdf}\\[-1pt]
      {\scriptsize Relative contrast $(D_{max}-D_{min})/D_{min}$ from random query points
      (Beyer et al.\ 1999).}
    \end{center}
  \end{columns}
\end{frame}

\section{PCA: the idea and the math}
\sectiontransition{Principal Component Analysis}{The linear workhorse: new axes along the
directions of greatest variance.}

\begin{frame}{Why maximize \emph{variance}?}
  Projecting always loses information: two points that differ only along a \emph{dropped}
  direction land on \textbf{the same point}. So which directions can we afford to lose?
  \begin{center}
    \includegraphics[width=0.80\textwidth]{fig/dr_why_variance.pdf}
  \end{center}
  \vskip -3pt
  \footnotesize Variance measures how much the samples \textbf{disagree} along an axis. Keep the
  spread axis and A and B stay clearly apart; keep the flat one and they nearly collide. An axis
  with zero variance --- the same value for every sample --- can be deleted free of charge.
  \textbf{PCA's bet: spread $=$ information.}
\end{frame}

\begin{frame}{PCA: keep the directions of greatest spread}
  Sweep a direction through the cloud and \textbf{project} the points onto it (red dots). The
  \textbf{projected variance} $\bm{w}^\top\bm{\Sigma}\bm{w}$ --- how spread out those red dots are ---
  rises and falls, peaking at the \textbf{principal axis}. Keep the \textbf{top-$k$} such axes
  (here 2-D$\to$1-D; for the 784-D garments we will keep 2).
  \begin{center}
    \only<1>{\includegraphics[width=0.50\textwidth]{fig/dr_pca_anim_1.pdf}}%
    \only<2>{\includegraphics[width=0.50\textwidth]{fig/dr_pca_anim_2.pdf}}%
    \only<3>{\includegraphics[width=0.50\textwidth]{fig/dr_pca_anim_3.pdf}}%
    \only<4>{\includegraphics[width=0.50\textwidth]{fig/dr_pca_anim_4.pdf}}%
    \only<5>{\includegraphics[width=0.50\textwidth]{fig/dr_pca_anim_5.pdf}}%
  \end{center}
  \vskip -2pt
  \footnotesize \textbf{PCA does not search or iterate} --- the eigenvectors (next frames) give
  these axes directly; the sweep just shows what ``maximum variance'' means.
\end{frame}

\begin{frame}{Refresher: the covariance matrix}
  \begin{columns}[T]
    \column{0.52\textwidth}
    For \textbf{centered} data (mean $= 0$):
    \[ \bm{\Sigma} = \tfrac{1}{n}\textstyle\sum_i \xv_i \xv_i^\top = \tfrac{1}{n}X^\top X, \]
    \vskip -8pt
    \[ \Sigma_{jk} = \tfrac{1}{n}\textstyle\sum_i x_{ij}\,x_{ik}. \]
    How to read it:
    \begin{itemize}\small
      \item \textbf{diagonal} $\Sigma_{jj}$ --- the \textbf{variance} of feature $j$;
      \item \textbf{off-diagonal} $\Sigma_{jk}$ --- do features $j$ and $k$ move
        \textbf{together} ($>0$), \textbf{opposite} ($<0$), or independently ($\approx 0$);
      \item \textbf{symmetric}: $\Sigma_{jk}=\Sigma_{kj}$ --- remember this for the next frame.
    \end{itemize}
    {\scriptsize (Software divides by $n-1$ instead of $n$; nothing below depends on which.)}
    \column{0.46\textwidth}
    \textbf{The one identity PCA runs on.} Project every point onto a unit direction $\bm{w}$:
    the projected values are $\xv_i^\top\bm{w}$, and their variance is
    \begin{align*}
      \mathrm{Var}(X\bm{w}) &= \tfrac{1}{n}\textstyle\sum_i (\xv_i^\top\bm{w})^2 \\
      &= \bm{w}^\top\big(\tfrac{1}{n}\textstyle\sum_i \xv_i\xv_i^\top\big)\bm{w}
       = \bm{w}^\top\bm{\Sigma}\,\bm{w}.
    \end{align*}
    \vskip 2pt
    \footnotesize One matrix, computed \emph{once}, gives the projected variance in
    \emph{every} direction at once --- which is why the sweep on the previous frame never had
    to touch the raw data twice.
  \end{columns}
\end{frame}

\begin{frame}{Refresher: eigenvectors and eigenvalues}
  \begin{columns}[T]
    \column{0.54\textwidth}
    An \textbf{eigenvector} $\bm{v}$ of a matrix $A$ keeps its \emph{direction} when $A$ acts on it;
    it only gets scaled by the \textbf{eigenvalue} $\lambda$:
    \[ A\bm{v} = \lambda \bm{v}. \]
    \vskip 4pt
    A \textbf{symmetric} matrix --- like a covariance --- has real eigenvalues and
    \textbf{orthogonal} eigenvectors. \emph{That is why the principal axes come out perpendicular.}
    \column{0.46\textwidth}
    \begin{center}
      \begin{tikzpicture}[scale=0.95, >=stealth]
        \draw[->, gray] (-0.3,0) -- (3.4,0);
        \draw[->, gray] (0,-0.3) -- (0,2.6);
        \draw[->, armblue, very thick] (0,0) -- (1.4,0.9) node[above left]{$\bm{v}$};
        \draw[->, armred, very thick] (0,0) -- (2.8,1.8) node[right]{$A\bm{v}=\lambda\bm{v}$};
        \node[armred, font=\scriptsize] at (2.0,0.7) {same direction, $\times\lambda$};
      \end{tikzpicture}
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}{Derivation (1): maximize variance}
  \textbf{Step 0 --- center the data} (subtract each feature's mean). Without it, $X^\top X$ measures
  spread about the \emph{origin}, not about the data.
  \vfill
  Variance of the data projected onto a unit direction $\bm{w}$ is $\bm{w}^\top \bm{\Sigma}\bm{w}$
  ($\bm{\Sigma}=$ covariance of the centered data). Maximize it subject to $\|\bm{w}\|=1$.
  Form the \textbf{Lagrangian} and set its gradient to zero:
  \[
    \mathcal{L}(\bm{w},\lambda) = \bm{w}^\top\bm{\Sigma}\bm{w} - \lambda(\bm{w}^\top\bm{w}-1),
    \qquad
    \nabla_{\bm{w}}\mathcal{L} = 2\bm{\Sigma}\bm{w} - 2\lambda\bm{w} = 0
  \]
  \[
    \;\;\Longrightarrow\;\; \bm{\Sigma}\bm{w} = \lambda\,\bm{w}.
  \]
  \vfill
  \fcolorbox{armblue}{armblue!8}{\parbox{0.97\linewidth}{\small
    The principal directions are the \textbf{eigenvectors of the covariance matrix}; each
    eigenvalue $\lambda$ is the \textbf{variance captured} along that direction.}}
\end{frame}

\begin{frame}{Derivation (2): explained variance}
  \begin{itemize}\small
    \item Eigenvalues sorted \textbf{descending} $\Rightarrow$ principal components come out
      \textbf{ordered by variance}; take the top $k$ eigenvectors.
    \item \textbf{Explained-variance ratio (EVR)} of component $i$ is $\lambda_i / \sum_j \lambda_j$
      (first PC of the garments $\approx 29\%$; first two $\approx 47\%$).
    \item The scores are \textbf{uncorrelated}: $\mathrm{Cov}(Z) = W^\top\bm{\Sigma}W =
      \mathrm{diag}(\lambda_1,\dots,\lambda_k)$. This is the precise sense in which PCA
      \emph{decorrelates} --- not that it ``removes correlated features''.
  \end{itemize}
  \vfill
  \fcolorbox{paramgreen}{paramgreen!8}{\parbox{0.97\linewidth}{\small
    \textbf{Two views, one answer.} Maximizing variance \textbf{is} minimizing
    \textbf{reconstruction error}: projecting onto the top-$k$ axes and lifting back loses the least.
    That is the basis for compression and denoising later.}}
\end{frame}

\begin{frame}{What PCA does to a point --- and what ``linear'' means}
  Stack the top-$k$ eigenvectors as the columns of $W_k$. Every point is mapped by the
  \textbf{same matrix}:
  \[ \bm{z} = W_k^\top(\xv - \bar{\xv}), \qquad z_i = \bm{v}_i^\top(\xv - \bar{\xv}), \]
  so each new feature $z_i$ is a \textbf{fixed weighted sum} of the original features --- one
  recipe, applied identically everywhere in space. That is what \textbf{linear} means.
  \begin{center}
    \includegraphics[width=0.82\textwidth]{fig/dr_linear_grid.pdf}
  \end{center}
  \vskip -3pt
  \footnotesize A linear map can \textbf{rotate}, \textbf{stretch}, \textbf{shear} and
  \textbf{project} (a shadow) --- grid lines stay straight, parallel stays parallel. What it can
  \emph{never} do is \textbf{bend}. Hold that thought for the nonlinear section.
\end{frame}

\begin{frame}{The SVD connection}
  In practice PCA is computed by the \textbf{singular value decomposition (SVD)} of the centered
  data --- and every factor has a PCA meaning:
  \begin{center}
    \begin{tikzpicture}[every node/.style={font=\scriptsize}, yscale=0.82, xscale=1.12]
      \draw[thick, popblue, fill=armblue!8] (0,0) rectangle (1.1,2.2);
      \node[font=\small] at (0.55,1.1) {$X_{\text{c}}$};
      \node[below, align=center] at (0.55,-0.08) {centered data\\$n \times d$};
      \node[font=\small] at (1.6,1.1) {$=$};
      \draw[thick, paramgreen, fill=paramgreen!8] (2.1,0) rectangle (3.2,2.2);
      \node[font=\small] at (2.65,1.1) {$U$};
      \node[below, align=center] at (2.65,-0.08) {$n \times d$\\one row per sample:\\its coordinates on\\the new axes};
      \draw[thick, armorange, fill=armorange!12] (3.7,1.1) rectangle (4.8,2.2);
      \draw[armorange, thick] (3.7,2.2) -- (4.8,1.1);
      \node[font=\small] at (4.45,1.95) {$S$};
      \node[below, align=center] at (4.25,0.82) {$d \times d$ diagonal\\$\sigma_1 \ge \sigma_2 \ge \dots$\\(the spreads)};
      \draw[thick, armred, fill=armred!8] (6.1,1.1) rectangle (7.2,2.2);
      \node[font=\small] at (6.65,1.65) {$V^\top$};
      \node[below, align=center] at (6.65,0.82) {$d \times d$\\rows $=$ the principal\\directions $\bm{v}_i$};
    \end{tikzpicture}
  \end{center}
  \begin{itemize}\small
    \item $V$ holds the \textbf{same eigenvectors} as the derivation, computed \emph{without}
      ever forming $\bm{\Sigma}$ --- more numerically stable.
    \item $S$ is sorted, and $\lambda_i \propto \sigma_i^2$: the scree plot reads straight off
      the diagonal. The \textbf{scores} are $Z = US$ --- what \texttt{fit\_transform} returns.
    \item \textbf{TruncatedSVD} (a.k.a.\ \emph{latent semantic analysis, LSA}, for text) skips
      centering, so it works on \textbf{sparse} data --- centering would destroy the sparsity.
  \end{itemize}
\end{frame}

\begin{frame}{PCA by hand (2-D)}
  \begin{columns}[T]

    \footnotesize
    \begin{enumerate}
      \item Center the points (subtract the mean, the \textbf{X}).
      \item Form the $2\times2$ \textbf{covariance} matrix.
      \item Its \textbf{eigenvectors} are the principal axes; \textbf{eigenvalues} $\lambda$ are the

\section{PCA in practice}
\sectiontransition{PCA in practice}{Scaling, how many components to keep, what they mean, and where
PCA bites.}

\begin{frame}{Centering vs scaling}
  \begin{itemize}\small
    \item PCA \textbf{always centers} (scikit-learn does it for you) --- mandatory.
    \item PCA does \textbf{not standardize} --- that is your choice.
  \end{itemize}
  \vskip -2pt
  \begin{center}
    \includegraphics[width=0.80\textwidth]{fig/dr_scaling_trap.pdf}
  \end{center}
  \vskip -4pt
  \fcolorbox{armred}{armred!8}{\parbox{0.97\linewidth}{\scriptsize
    \textbf{Trap:} variance is scale-dependent. In dollars, PC1 is $(1.00,\,0.00)$ --- literally
    the income axis, and age contributes nothing. Standardized, PC1 is $(0.71,\,0.71)$: the real
    shared direction. \textbf{Standardize} unless your features already share units (like pixel
    intensities, which is why we do not standardize the garments).}}
\end{frame}

\begin{frame}{How many components to keep?}
  \textbf{Predict-first:} the garments are 784-D and the first PC alone is 29\%. How many
  components do you need to reach \textbf{95\%}?
  \pause
  \begin{center}
    \includegraphics[width=0.87\textwidth]{fig/dr_scree.pdf}
  \end{center}
  \vskip -4pt
  \footnotesize \textbf{184} --- almost a quarter of them, not a handful. \textbf{Scree} (left):
  variance per component --- read its elbow like k-means' inertia elbow. \textbf{Cumulative}
  (right): the tail is long and flat, and that is where the detail lives.
  \vskip 2pt
  {\scriptsize \textbf{When PCA feeds a model}: do not pick $k$ by a variance threshold --- tune
  it by CV on the \emph{downstream} metric, with the PCA fitted \emph{inside} the pipeline on
  training folds only (the leakage rule again).}
\end{frame}

\begin{frame}{Reading a biplot}
  \begin{columns}[T]
    \column{0.5\textwidth}
    A \textbf{biplot} shows two things on the same PC1--PC2 axes:
    \begin{itemize}\small
      \item \textbf{dots} $=$ the samples projected onto PC1/PC2 (colored by species here);
      \item \textbf{arrows} $=$ the original features (their \emph{loadings}).
    \end{itemize}
    How to read an arrow:
    \begin{itemize}\small
      \item \textbf{direction} $=$ the way that feature increases;
      \item \textbf{length} $=$ how strongly it drives these two PCs;
      \item arrows pointing the \textbf{same way} $=$ correlated features.
    \end{itemize}
    \footnotesize So petal length \& width point together (correlated) and separate the species along
    PC1; sepal width points elsewhere. \emph{Components are combinations --- only partly interpretable.}
    \column{0.5\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig/dr_biplot.pdf}
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}{The principal directions, drawn as images}
  Each $\bm{v}_i$ lives in the \emph{same} 784-dimensional pixel space as the garments --- so
  every component \textbf{is} a $28\times28$ image, and we can simply look at it.
  \begin{center}
    \includegraphics[width=0.97\textwidth]{fig/dr_eigengarments.pdf}
  \end{center}
  \vskip -3pt
  \footnotesize \textcolor{armred}{Red} and \textcolor{armblue}{blue} $=$ pixels the component
  pushes in \emph{opposite} directions; a sample's score $z_i$ says how much of the pattern to
  mix in, with sign. PC1 ($29\%$) is roughly ``how much of the frame the garment fills'';
  PC2 contrasts trouser-like verticals with broad tops; the later, smaller components draw
  ever finer shape details.
  \vskip 2pt
  \fcolorbox{armblue}{armblue!8}{\parbox{0.97\linewidth}{\small
    Reconstruction (next frame) is literally: start from the mean garment and \textbf{mix in}
    $z_i$ units of each of these pictures. In Project 1 you will draw these for faces --- the
    famous \emph{eigenfaces}.}}
\end{frame}

\begin{frame}{Reconstruction: compression and denoising}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \footnotesize
    \textbf{Project} to $k$ scores, then \textbf{lift back} to the original space:
    \[ \bm{z} = W_k^\top(\xv-\bar{\xv}), \qquad \hat{\xv} = \bar{\xv} + W_k\,\bm{z}. \]
    ``Lift back'' $=$ the \textbf{mean plus a weighted sum of the kept axes}:
    \[ \hat{\xv} = \bar{\xv} + \textstyle\sum_{i=1}^{k} z_i\,\bm{v}_i \]
    ($\bm{v}_i$ the principal directions, $z_i$ the scores). Fewer $k$ $=$ more compression (blurrier).
    \vskip 3pt
    784 numbers $\to$ \textbf{50} and the garment is still clearly itself.
    \vskip 3pt
    \textbf{Callback:} the blurriness at $k{=}5$ is exactly the variance the scree plot said we
    threw away. The squared reconstruction error is $\sum_{i>k}\lambda_i$ (the dropped
    eigenvalues); as a \emph{fraction} of the total, that is $1-$ cumulative EVR.
    \vskip 3pt
    Noise lives in the small-variance directions you dropped, so the rebuild is also \textbf{denoised}.
    \column{0.54\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig/dr_reconstruction.pdf}
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}{PCA pitfalls}
  \begin{itemize}\small
    \item \textbf{Not feature selection} --- components are \emph{combinations} of all features, not a
      chosen subset. DR is feature \textbf{extraction} (build new features); the other family,
      feature \textbf{selection}, keeps a subset of the originals (drop low-variance or correlated
      columns, L1, \dots). Extraction gives you better axes; selection gives you back your
      original, readable ones.
    \item \textbf{Linear only} --- it cannot unfold a curved manifold (next section).
    \item \textbf{Sensitive} to outliers and to feature scale.
    \item Components can be \textbf{hard to interpret}.
  \end{itemize}
\end{frame}

\section{Nonlinear DR for visualization: t-SNE \& UMAP}
\sectiontransition{Nonlinear methods}{When the structure is a curved manifold, linear PCA is not
enough --- t-SNE and UMAP preserve \emph{local} neighborhoods.}

\begin{frame}{Why nonlinear?}
  PCA is \textbf{linear}: remember the grid --- it can rotate, stretch and project, but it can
  never \emph{bend}, so it cannot \emph{unfold} a curved manifold. Nonlinear methods instead
  preserve \textbf{local neighborhoods} --- who is near whom.
  \begin{center}
    \includegraphics[width=0.94\textwidth]{fig/dr_swiss_roll.pdf}
  \end{center}
  \vskip -4pt
  \scriptsize Color $=$ position along the roll. \textbf{PCA} gives you a \emph{shadow}: the roll
  stays rolled, so points on \emph{adjacent turns} sit side by side even though they are far
  apart along the sheet. \textbf{UMAP} sweeps the colors once, end to end. It did not lay the
  sheet out straight --- it had no reason to. Topology preserved, geometry not.
\end{frame}

\begin{frame}{t-SNE: t-distributed Stochastic Neighbor Embedding}
  \begin{columns}[T]
    \column{0.52\textwidth}
    Turn distances into \textbf{similarity probabilities}, high-D and low-D, then match them.
    \begin{itemize}\small
      \item High-D: $p_{ij}=\tfrac{1}{2n}(p_{j|i}+p_{i|j})$ (Gaussian neighborhoods).
      \item Low-D: $q_{ij}$ from a \textbf{Student-$t$} kernel; its heavy tail fixes
        \emph{crowding} --- 2-D lacks room for all the mid-distance neighbors, so they may
        sit farther out penalty-free.
      \item Minimize $\mathrm{KL}(P\,\|\,Q)$ by gradient descent (iterative, \emph{stochastic}).
    \end{itemize}
    \textbf{Perplexity} $\approx$ effective \#neighbors (5--50 typ.).
    \vskip 3pt
    {\scriptsize Sandal, sneaker, ankle boot are each other's nearest classes here --- so are
    pullover, coat, shirt. \textbf{t-SNE recovered the semantics from raw pixels}, no labels.}
    \column{0.48\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig/dr_tsne_fashion.pdf}
    \end{center}
  \end{columns}
  \vskip -4pt
  \hfill {\scriptsize van der Maaten \& Hinton (2008).}
\end{frame}

\begin{frame}{UMAP}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \textbf{Uniform Manifold Approximation and Projection (UMAP)}: build a \textbf{fuzzy
    nearest-neighbor graph}, then find a low-D layout whose graph matches it (minimize a graph
    \textbf{cross-entropy}).
    \begin{itemize}\small
      \item \texttt{n\_neighbors}: small $=$ local detail, large $=$ global shape.
      \item \texttt{min\_dist}: how tightly points may pack.
      \item \textbf{Faster} than t-SNE, scales better, keeps \emph{more} global structure.
    \end{itemize}
    \vskip 2pt
    \fcolorbox{paramgreen}{paramgreen!8}{\parbox{0.95\linewidth}{\scriptsize
      This frame is \emph{what UMAP gives you}. \textbf{How it works} --- the fuzzy graph,
      $\rho_i$ and $\sigma_i$, the cross-entropy --- is the whole of the next lecture.}}
    \column{0.48\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig/dr_umap_fashion.pdf}
    \end{center}
  \end{columns}
  \vskip -2pt
  \hfill {\scriptsize McInnes et al.\ (2018).}
\end{frame}

\begin{frame}{Caveats: t-SNE / UMAP can mislead}
  \textbf{Predict-first:} same data, same algorithm, one hyperparameter moved. Will the
  picture be the same?
  \pause
  \vskip -2pt
  \begin{center}
    \includegraphics[width=0.90\textwidth]{fig/dr_tsne_perplexity.pdf}
  \end{center}
  \vskip -5pt
  \fcolorbox{armred}{armred!8}{\parbox{0.985\linewidth}{\scriptsize
    So: \textbf{cluster sizes and between-cluster distances are not meaningful}; results are
    \textbf{stochastic}; \textbf{do not feed the 2-D embedding to a downstream model} --- it is
    for the eyes. {\normalfont (Wattenberg et al., distill.pub 2016.)}}}
\end{frame}

\section{Choosing a method}
\sectiontransition{Which one?}{PCA, t-SNE, UMAP --- side by side, then a decision table.}

\begin{frame}{PCA vs t-SNE vs UMAP on the garments}
  \begin{center}
    \includegraphics[width=0.97\textwidth]{fig/dr_compare_fashion.pdf}
  \end{center}
  \vskip -2pt
  \footnotesize PCA (linear) piles the classes on top of each other; t-SNE and UMAP pull them
  apart. PCA is for preprocessing; t-SNE / UMAP are for \emph{looking}.
  \vskip 2pt
  {\scriptsize \textbf{In practice} the nonlinear methods here were run on the top 50 PCs, not the
  raw 784 pixels: PCA first as a denoiser and accelerator, then t-SNE / UMAP for the picture.
  That stack is the normal recipe, not a shortcut.}
\end{frame}

\begin{frame}{Decision table}
  \begin{center}\footnotesize
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{l c c c c l}
      & \textbf{Linear?} & \textbf{Global?} & \textbf{Determin.?} & \textbf{New pts?} & \textbf{Use it for} \\
      \hline
      \textbf{PCA} & yes & yes & yes & yes & preprocessing, compression, denoising \\
      \textbf{t-SNE} & no & no & no & no & visualizing local cluster structure \\
      \textbf{UMAP} & no & some & no (seeded) & approx. & visualization, faster / larger data \\
    \end{tabular}
  \end{center}
  \vskip 4pt
  {\scriptsize \textbf{New points?} --- can a fitted model embed data it has never seen?
  \textbf{PCA}: \texttt{transform}, exact. \textbf{UMAP}: \texttt{transform}, approximate.
  \textbf{t-SNE}: no --- it learns positions, not a map, so a new point means refitting
  everything. One more reason the 2-D picture cannot be a feature for a model.}
  \vfill
  \footnotesize \textbf{Rule of thumb:} reach for \textbf{PCA} first (fast, cheap, reversible); use
  \textbf{t-SNE / UMAP} when you specifically want a 2-D \emph{picture} of the structure.
\end{frame}

\begin{frame}{When NOT to reduce}
  \fcolorbox{armorange}{armorange!12}{\parbox{0.97\linewidth}{\small
    DR throws information away --- so do not reduce reflexively:}}
  \begin{itemize}\small
    \item PCA is \textbf{unsupervised}: it ranks directions by variance \emph{with no knowledge of
      $y$}. The direction that separates your labels can sit in a \textbf{low-variance} component PCA
      discards (a thin signal axis vs.\ a fat nuisance axis).
    \item \textbf{Tree models} (forests, boosting) usually do not need it.
    \item Always try the model \emph{without} DR first; add it only if it helps.
    \item And if the goal is separating \emph{classes}: there is a supervised cousin,
      \textbf{Linear Discriminant Analysis (LDA)}, which picks directions by class separation
      rather than variance. You will meet it as \emph{Fisherfaces} in Project 2.
  \end{itemize}
\end{frame}

\section*{Wrap-up}

\begin{frame}{Recap}
  \begin{itemize}\small
    \item \textbf{Why:} visualize, compress, denoise --- and beat the curse of dimensionality
      (in 100-D your farthest point is only $\sim$40\% farther than your nearest).
    \item \textbf{PCA} --- center, then take the top eigenvectors of the covariance (via SVD);
      the scores come out uncorrelated and ordered by variance; linear, fast, reversible.
      Great for preprocessing. On the garments, 95\% of the variance needs \textbf{184 of 784}
      components --- but 50 already rebuilds a recognizable shirt.
    \item \textbf{t-SNE / UMAP} --- nonlinear, preserve local neighborhoods, excellent for 2-D
      \emph{pictures} --- but sizes/distances are not meaningful and they are for the eyes only.
    \item Reach for PCA first; don't reduce reflexively (it is unsupervised, and it loses information).
  \end{itemize}
  \vfill
  \fcolorbox{paramgreen}{paramgreen!8}{\parbox{0.97\linewidth}{\small
    \textbf{Next (lecture 36):} UMAP, opened up --- the fuzzy neighbor graph, why each point gets its
    own distance scale, and what the cross-entropy is really doing. After that, unsupervised
    learning is closed and we move to a new family of models.}}
\end{frame}
